//@version=5
indicator("-Quick Trade Indicator-(QTI)-(RSI+Vortex+Alligator)-", shorttitle="-Quick Trade Indicator-(QTI)-(RSI+Vortex+Alligator)-", overlay=true)

// Inputparameter für RSI
rsi_source = input.source(close, title="-RSI-Source-close-")
rsi_period = input.int(14, title="-RSI-Period-14-")
rsi_ob_level = input.int(70, title="-RSI-Overbought-Level-70-")
rsi_os_level = input.int(30, title="-RSI-Oversold-Level-30-")

// Inputparameter für Vortex
vortex_period = input.int(14, title="-Vortex-Period-14-")

// Inputparameter für Alligator
jaw_source = input.source(close, title="-Alligator-Jaw-Source-close-")
jaw_length = input.int(13, title="-Alligator-Jaw-Length-13-")
teeth_source = input.source(close, title="-Alligator-Teeth-Source-close-")
teeth_length = input.int(8, title="-Alligator-Teeth-Length-8-")
lips_source = input.source(close, title="-Alligator-Lips-Source-close-")
lips_length = input.int(5, title="-Alligator-Lips-Length-5-")

// Berechnung des RSI
rsi = ta.rsi(rsi_source, rsi_period)

// Custom Sum Function
f_sum(src, length) =>
    float result = 0.0
    for i = 0 to length - 1
        result := result + src[i]
    result

// Berechnung des Vortex-Indikators
tr = ta.tr(true)
vm_plus = f_sum(math.abs(high - low[1]), vortex_period)
vm_minus = f_sum(math.abs(low - high[1]), vortex_period)
vi_plus = vm_plus / f_sum(tr, vortex_period)
vi_minus = vm_minus / f_sum(tr, vortex_period)

// Berechnung des Alligator-Indikators
jaw = ta.sma(jaw_source, jaw_length)
teeth = ta.sma(teeth_source, teeth_length)
lips = ta.sma(lips_source, lips_length)

// Signale
buy_signal = (rsi < rsi_os_level) and (vi_plus > vi_minus) and ta.crossover(lips, teeth)
sell_signal = (rsi > rsi_ob_level) and (vi_plus < vi_minus) and ta.crossunder(lips, teeth)

// Plots
plot(jaw, color=color.blue, title="-Alligator-Jaw-", linewidth=2)
plot(teeth, color=color.red, title="-Alligator-Teeth-", linewidth=2)
plot(lips, color=color.green, title="-Alligator-Lips-", linewidth=2)

// Buy and Sell Markierungen
plotshape(series=buy_signal, title="-Buy-Signal-", location=location.belowbar, color=color.green, textcolor=color.black, style=shape.labelup, text="BUY", size=size.tiny)
plotshape(series=sell_signal, title="-Sell-Signal-", location=location.abovebar, color=color.red, textcolor=color.black, style=shape.labeldown, text="SELL", size=size.tiny)

// Zusatzplot für RSI und Vortex-Indikator
hline(rsi_ob_level, title="-RSI-Overbought-", color=color.red, linewidth=3, linestyle=hline.style_solid, display=display.none)
hline(rsi_os_level, title="-RSI-Oversold-", color=color.green, linewidth=3, linestyle=hline.style_solid, display=display.none)
plot(rsi, title="-RSI-", color=color.yellow, linewidth=2, display=display.none)
plot(vi_plus, title="-Vortex- +", color=color.green, linewidth=2, display=display.none)
plot(vi_minus, title="-Vortex- -", color=color.red, linewidth=2, display=display.none)
