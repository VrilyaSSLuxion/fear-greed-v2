//@version=5
indicator("-VrilyaSS-Day Trader-Candlestick Chart Signals-X18-+AO-", overlay=true)

// Helper function to identify bullish/bearish candles
isBullish(o, c) =>
    c > o

isBearish(o, c) =>
    c < o

// Helper function to find the midpoint of two prices
midpoint(price1, price2) =>
    (price1 + price2) / 2

// Function to check for a downtrend
isDowntrend() =>
    close[1] < close[2] and close[2] < close[3]

// Function to check for an uptrend
isUptrend() =>
    close[1] > close[2] and close[2] > close[3]

// Calculate Awesome Oscillator
aoLength1 = input.int(5, title="Awesome Oscillator Short Length")
aoLength2 = input.int(34, title="Awesome Oscillator Long Length")

// Calculate Awesome Oscillator values
midPrice = (high + low) / 2
ao = ta.sma(midPrice, aoLength1) - ta.sma(midPrice, aoLength2)

// AO confirmation conditions
aoBullish = ao > 0
aoBearish = ao < 0

// Bullish Patterns with Awesome Oscillator Confirmation
bullishHammer = (low - math.min(open, close) > 2 * (math.max(open, close) - high)) and isDowntrend() and aoBullish
bullishInvertedHammer = (high - math.max(open, close) > 2 * (math.min(open, close) - low)) and isDowntrend() and aoBullish
bullishPiercing = (isBearish(close[1], open[1]) and isBullish(open, close) and open < close[1] and close > midpoint(open[1], close[1])) and isDowntrend() and aoBullish
bullishHarami = (isBearish(close[1], open[1]) and isBullish(open, close) and open > close[1] and close < open[1]) and isDowntrend() and aoBullish
bullishKicking = (isBearish(close[1], open[1]) and isBullish(open, close) and low > high[1]) and isDowntrend() and aoBullish
bullishEngulfing = (isBearish(close[1], open[1]) and isBullish(open, close) and open < close[1] and close > open[1]) and isDowntrend() and aoBullish
bullishMorningStar = (isBearish(close[2], open[2]) and (open[1] < close[2] and close[1] > open[2]) and isBullish(open, close) and close > midpoint(open[2], close[2])) and isDowntrend() and aoBullish
bullishBeltHold = (isBullish(open, close) and low == open and close > midpoint(high, low)) and isDowntrend() and aoBullish
bullishThreeWhiteSoldiers = (isBullish(open[2], close[2]) and isBullish(open[1], close[1]) and isBullish(open, close) and close[1] > close[2] and close > close[1]) and isDowntrend() and aoBullish

// Bearish Patterns with Awesome Oscillator Confirmation
bearishHangingMan = (high - math.max(open, close) > 2 * (math.min(open, close) - low)) and isUptrend() and aoBearish
bearishShootingStar = (high - close > 2 * (open - low)) and isUptrend() and aoBearish
bearishBeltHold = (isBearish(open, close) and high == open and close < midpoint(high, low)) and isUptrend() and aoBearish
bearishHarami = (isBullish(close[1], open[1]) and isBearish(open, close) and open < close[1] and close > open[1]) and isUptrend() and aoBearish
bearishKicking = (isBullish(close[1], open[1]) and isBearish(open, close) and high < low[1]) and isUptrend() and aoBearish
bearishEngulfing = (isBullish(close[1], open[1]) and isBearish(open, close) and open > close[1] and close < open[1]) and isUptrend() and aoBearish
bearishEveningStar = (isBullish(close[2], open[2]) and (open[1] > close[2] and close[1] < open[2]) and isBearish(open, close) and close < midpoint(open[2], close[2])) and isUptrend() and aoBearish
bearishDarkCloud = (isBullish(close[1], open[1]) and isBearish(open, close) and open > close[1] and close < midpoint(open[1], close[1])) and isUptrend() and aoBearish
bearishThreeBlackCrows = (isBearish(open[2], close[2]) and isBearish(open[1], close[1]) and isBearish(open, close) and close[1] < close[2] and close < close[1]) and isUptrend() and aoBearish

// Plot Bullish Patterns
