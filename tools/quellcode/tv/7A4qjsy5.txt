//@version=5
indicator("Buy Low Sell High with Swing and Trend Filter", overlay=true)

// Parameter für die Anzahl der Kerzen, auf denen die extremen Werte basieren
length = input.int(14, title="Lookback Period", minval=1)
swingLength = input.int(5, title="Swing Length", minval=1)  // Länge für die Swing High/Low-Erkennung
smaLength = input.int(50, title="SMA Length", minval=1)  // Länge des gleitenden Durchschnitts
swingIndex = input.int(5, title="Swing Index", minval=0)  // Index für die Swing High/Low-Erkennung

// Berechnung der niedrigsten und höchsten Kurse der letzten `length` Kerzen
lowestLow = ta.lowest(low, length)
highestHigh = ta.highest(high, length)

// Berechnung des gleitenden Durchschnitts
sma = ta.sma(close, smaLength)

// Funktionen zur Erkennung von Swing Highs und Swing Lows
isSwingHigh(index) =>
    high[index] == ta.highest(high, swingLength * 2 + 1) and high[index] != ta.highest(high, swingLength)

isSwingLow(index) =>
    low[index] == ta.lowest(low, swingLength * 2 + 1) and low[index] != ta.lowest(low, swingLength)

// Identifikation von Swing Highs und Swing Lows unter Verwendung des variablen Swing Index
swingHigh = isSwingHigh(swingIndex)
swingLow = isSwingLow(swingIndex)

// Trendfilter: Überprüfen, ob der Preis über oder unter dem gleitenden Durchschnitt liegt
isBullishTrend = close > sma
isBearishTrend = close < sma

// Kauf- und Verkaufssignale basierend auf extremen Werten, Swing High/Low Filtern und Trendfilter
buySignal = low == lowestLow and swingLow and isBullishTrend
sellSignal = high == highestHigh and swingHigh and isBearishTrend

// Plotte die Kauf- und Verkaufssignale auf dem Chart
plotshape(series=buySignal, title="Buy Signal", location=location.belowbar, color=color.green, style=shape.labelup, text="Buy", textcolor=color.white, size=size.small)
plotshape(series=sellSignal, title="Sell Signal", location=location.abovebar, color=color.red, style=shape.labeldown, text="Sell", textcolor=color.white, size=size.small)

// Plotte den gleitenden Durchschnitt auf dem Chart
plot(sma, title="SMA", color=color.blue, linewidth=2)

// Plotte Swing Highs und Swing Lows auf dem Chart
plotshape(series=swingHigh, title="Swing High", location=location.abovebar, color=color.purple, style=shape.triangledown, text="SH", textcolor=color.white, size=size.small)
plotshape(series=swingLow, title="Swing Low", location=location.belowbar, color=color.orange, style=shape.triangleup, text="SL", textcolor=color.white, size=size.small)

// Alarme
alertcondition(buySignal, title="Buy Signal Alert", message="Buy signal generated: Price is at the lowest of the last period, a swing low is detected, and the trend is bullish.")
alertcondition(sellSignal, title="Sell Signal Alert", message="Sell signal generated: Price is at the highest of the last period, a swing high is detected, and the trend is bearish.")
