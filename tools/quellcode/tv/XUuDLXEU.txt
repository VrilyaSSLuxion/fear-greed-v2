//@version=5
indicator("-VrilyaSS-BB-(15-1)-Buy/Sell-Indicator-3-", overlay=true)

// Inputs für Bollinger Bands
bbLength = input.int(15, title="Bollinger Bands Length")
bbMultiplier = input.float(1.0, title="Bollinger Bands Multiplier")
[bbBasis, bbUpper, bbLower] = ta.bb(close, bbLength, bbMultiplier)

// Zustand des letzten Signals verfolgen
var bool signalGenerated = false
var int signalBarIndex = na

// Bedingungen für Verkaufsignal
currentCandleRed = close < open
closeBelowLowerBand = close < bbLower

// Bedingungen für Verkauf nach Umkehr
upperBandCrossed = ta.valuewhen(close > bbUpper, close, 0)
noContactWithBands = ta.lowest(close, 3) > bbLower and ta.highest(close, 3) < bbUpper
lowerHigh = high < ta.highest(high[1], 1)

// Zweites Verkaufsignal, wenn der Kurs über das obere Bollinger Band geht und die erste Kerze negativ schließt
sellSignal2Condition = ta.valuewhen(close > bbUpper, currentCandleRed, 0) and close < open

// Verkaufsignal generieren, wenn die Bedingungen erfüllt sind
sellSignalCondition = currentCandleRed and closeBelowLowerBand
sellReversalCondition = close > upperBandCrossed and noContactWithBands and lowerHigh and currentCandleRed

sellSignal = (sellSignalCondition or sellReversalCondition or sellSignal2Condition) and not signalGenerated

// Wenn ein Verkaufssignal generiert wird, setze signalGenerated auf true und speichere den Bar Index
if (sellSignal)
    signalGenerated := true
    signalBarIndex := bar_index

// Signal zurücksetzen, wenn der Kurs das obere Bollinger Band berührt hat und mindestens 3 Kerzen seit dem letzten Signal vergangen sind
if (high >= bbUpper and (bar_index - signalBarIndex >= 3))
    signalGenerated := false

// Bedingungen für Kaufsignal
currentCandleGreen = close > open
closeAboveLowerBand = close > bbLower
noTouchUpperBand = ta.highest(close, 3) <= bbUpper

// Bedingungen für Kauf nach Umkehr
lowerBandCrossed = ta.valuewhen(close < bbLower, close, 0)
noContactWithBandsBuy = ta.lowest(close, 3) > bbLower and ta.highest(close, 3) < bbUpper
higherLow = low > ta.lowest(low[1], 1)

// Kaufsignal generieren, wenn die Bedingungen erfüllt sind
buySignalCondition = currentCandleGreen and closeAboveLowerBand and noTouchUpperBand
buyReversalCondition = close < lowerBandCrossed and noContactWithBandsBuy and higherLow and currentCandleGreen

buySignal = (buySignalCondition or buyReversalCondition) and not signalGenerated

// Wenn ein Kaufsignal generiert wird, setze signalGenerated auf true und speichere den Bar Index
if (buySignal)
    signalGenerated := true
    signalBarIndex := bar_index

// Signal zurücksetzen, wenn der Kurs das untere Bollinger Band berührt hat und mindestens 3 Kerzen seit dem letzten Signal vergangen sind
if (low <= bbLower and (bar_index - signalBarIndex >= 3))
    signalGenerated := false

// Plotte Bollinger Bands
plot(bbUpper, color=color.red, title="Bollinger Upper Band")
plot(bbBasis, color=color.blue, title="Bollinger Basis")
plot(bbLower, color=color.green, title="Bollinger Lower Band")

// Plotte Verkaufsignal
plotshape(series=sellSignal, title="Sell Signal", location=location.abovebar, color=color.red, style=shape.labeldown, text="Sell", textcolor=color.white, size=size.small)

// Plotte Kaufsignal
plotshape(series=buySignal, title="Buy Signal", location=location.belowbar, color=color.green, style=shape.labelup, text="Buy", textcolor=color.white, size=size.small)

// Alarme
alertcondition(sellSignal, title="Sell Signal Alert", message="Sell signal generated based on Bollinger Bands and candle conditions.")
alertcondition(buySignal, title="Buy Signal Alert", message="Buy signal generated based on Bollinger Bands and candle conditions.")
