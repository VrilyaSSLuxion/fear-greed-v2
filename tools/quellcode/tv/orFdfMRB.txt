//@version=5
indicator(title="-VrilyaSS-Tyler Jenks-Concensio System-1+2+3-", shorttitle="-VrilyaSS-Tyler Jenks-Concensio System-1+2+3-", overlay=true)

// Define SMAs
sma3 = ta.sma(close, 3)
sma10 = ta.sma(close, 10)
sma30 = ta.sma(close, 30)

// Plot SMAs
plot(sma3, title="-SMA3-", color=color.yellow, linewidth=2)
plot(sma10, title="-SMA10-", color=color.green, linewidth=3)
plot(sma30, title="-SMA30-", color=color.red, linewidth=4)

// Buy and Sell signals based on all specified conditions
buy_signal1 = ta.crossover(sma3, sma10) and sma3 > sma30
sell_signal1 = ta.crossunder(sma3, sma10) and sma3 < sma30

buy_signal2 = ta.crossover(sma3, sma10) and sma3 < sma30
sell_signal2 = ta.crossunder(sma3, sma10) and sma3 > sma30

buy_signal3 = ta.crossover(sma3, sma10) and sma3 < sma30 or ta.crossover(sma3, sma30)
sell_signal3 = ta.crossunder(sma3, sma10) and sma3 > sma30 or ta.crossunder(sma3, sma30)

buy_signal = buy_signal1 or buy_signal2 or buy_signal3
sell_signal = sell_signal1 or sell_signal2 or sell_signal3

// Plot buy and sell signals
plotshape(series=buy_signal ? sma3 : na, title="-Buy-Signal-", location=location.belowbar, color=color.green, textcolor=color.black, style=shape.labelup, text="Buy", size=size.tiny)
plotshape(series=sell_signal ? sma3 : na, title="-Sell-Signal-", location=location.abovebar, color=color.red, textcolor=color.black, style=shape.labeldown, text="Sell", size=size.tiny)

// Alerts
alertcondition(buy_signal, title="Buy Signal", message="Buy Signal -VrilyaSS-Tyler Jenks-Concensio System-")
alertcondition(sell_signal, title="Sell Signal", message="Sell Signal -VrilyaSS-Tyler Jenks-Concensio System-")
