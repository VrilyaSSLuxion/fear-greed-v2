//@version=5
indicator("-VrilyaSS-RSI + Stochastic Strategy-", overlay=true)

// Input parameters for Stochastic Oscillator
stoch_length = input.int(14, title="Stochastic Length")
stoch_k = input.int(3, title="%K Smoothing")
stoch_d = input.int(3, title="%D Smoothing")
stoch_overbought = input.int(80, title="Stochastic Overbought Level")
stoch_oversold = input.int(20, title="Stochastic Oversold Level")

// Input parameters for RSI
rsi_source = input.source(close, title="RSI Source")
rsi_length = input.int(14, title="RSI Length")
rsi_overbought = input.int(70, title="RSI Overbought Level")
rsi_oversold = input.int(30, title="RSI Oversold Level")

// Calculate Stochastic Oscillator
k = ta.sma(ta.stoch(close, high, low, stoch_length), stoch_k)
d = ta.sma(k, stoch_d)

// Calculate RSI
rsi = ta.rsi(rsi_source, rsi_length)

// Define Buy and Sell Conditions
buy_condition = (k < stoch_oversold and d < stoch_oversold) and (rsi < rsi_oversold)
sell_condition = (k > stoch_overbought and d > stoch_overbought) and (rsi > rsi_overbought)

// Plot Buy and Sell Signals
plotshape(series=buy_condition, location=location.belowbar, color=color.green, textcolor=color.black, style=shape.labelup, text="Buy", title="Buy Signal", size=size.small)
plotshape(series=sell_condition, location=location.abovebar, color=color.red, textcolor=color.black, style=shape.labeldown, text="Sell", title="Sell Signal", size=size.small)

// Alert Conditions
alertcondition(buy_condition, title="Buy Signal Alert", message="RSI + Stochastic Buy Signal")
alertcondition(sell_condition, title="Sell Signal Alert", message="RSI + Stochastic Sell Signal")

// General Buy/Sell Signal Alert
alertcondition(buy_condition or sell_condition, title="Buy/Sell Signal Alert", message="RSI + Stochastic Buy or Sell Signal")
