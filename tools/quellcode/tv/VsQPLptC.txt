indicator('-VrilyaSS-TSA TDI-TraderJe-Version#1-Conditions-V1-', shorttitle='-VrilyaSS-TSA TDI-TraderJe-Version#1-Conditions-V1-', format=format.price, precision=2)

// TDI
rsi = ta.rsi(close, 8)
ma = ta.sma(rsi, 34)
offs = 1.6185 * ta.stdev(rsi, 34)
up = ma + offs
dn = ma - offs
mid = (up + dn) / 2
fastMA = ta.sma(rsi, 2)

hline(79, '-OverBought-default-Color-gray-5-100%-Line Width-1-dashed Line-75-', color=color.gray, linestyle=hline.style_dashed, editable=true)
hline(21, '-OverSold-default-Color-gray-5-100%-Line Width-1-dashed Line-25-', color=color.gray, linestyle=hline.style_dashed, editable=true)

hline(89, '-Upper-Horizontal Line-default-Color-gray-5-100%-Line Width-1-dashed Line-80-', color=color.gray, linestyle=hline.style_dashed, editable=true)
hline(50, '-Middle-Horizontal Line-default-Color-gray-5-100%-Line Width-1-dashed Line-50-', color=color.white, linestyle=hline.style_dashed, linewidth=2, editable=true)
hline(11, '-Lower-Horizontal Line-default-Color-gray-5-100%-Line Width-1-dashed Line-20-', color=color.gray, linestyle=hline.style_dashed, editable=true)

upl = plot(up, '-Upper Band-', color=color.new(color.red, 0), linewidth=3)
dnl = plot(dn, '-Lower Band-', color=color.new(color.green, 0), linewidth=3)
midl = plot(mid, '-Middle of Bands-', color=color.new(color.white, 0), linewidth=2)

plot(fastMA, '-Fast MA-', color=color.new(color.yellow, 0), linewidth=2)

// Bedingungen für Kauf- und Verkaufssignale
buy_signal_condition = fastMA <= 30 and fastMA < mid and fastMA < dn
sell_signal_condition = fastMA >= 70 and fastMA > mid and fastMA > up

// Initialisiere Zähler für Signale
var int buy_signal_counter = 0
var int sell_signal_counter = 0

// Bestimme Kauf- und Verkaufssignale unter Berücksichtigung des Zählers
buy_signal = buy_signal_condition and buy_signal_counter == 0
sell_signal = sell_signal_condition and sell_signal_counter == 0

// Wenn Kaufsignal gepostet wurde, setze Zähler und Signal zurück
if buy_signal
    buy_signal_counter := 5

// Wenn Verkaufssignal gepostet wurde, setze Zähler und Signal zurück
if sell_signal
    sell_signal_counter := 5

// Verringere Zähler nach jedem Bar um 1
if buy_signal_counter > 0
    buy_signal_counter := buy_signal_counter - 1

if sell_signal_counter > 0
    sell_signal_counter := sell_signal_counter - 1

// Plot buy and sell labels at the relevant points
if (buy_signal)
    label.new(bar_index, dn, text="Buy", style=label.style_label_up, color=color.green, textcolor=color.black, yloc=yloc.price, size=size.small)
if (sell_signal)
    label.new(bar_index, up, text="Sell", style=label.style_label_down, color=color.red, textcolor=color.black, yloc=yloc.price, size=size.small)

// Alerts for buy and sell signals
alertcondition(buy_signal, title='Buy Alert', message='Buy Signal on {{ticker}}')
alertcondition(sell_signal, title='Sell Alert', message='Sell Signal on {{ticker}}')

// Bestehende Alarmbedingungen
finUp = fastMA > 75 and fastMA > up
finDn = fastMA < 25 and fastMA < dn
alert = finUp or finDn

if alert
    alert('{"content":"' + syminfo.ticker + ' ' + timeframe.period + ' Sharkfin"}', alert.freq_once_per_bar_close)
