//@version=5
indicator("-VrilyaSS-Binary Tricks Revn-5+10-SMA-Cross-15S-Chart-1MTrade-X9", shorttitle="-VrilyaSS-Binary Tricks Revn-5+10-SMA-Cross-15S-Chart-1MTrade-X9", overlay=true)

// Eingaben für die Fast SMA
fast_sma_source = input.source(close, title="-Fast SMA 1 Source")
fast_sma_length = input.int(5, title="-Fast SMA 1 Length-5-")

// Eingaben für die Slow SMA
slow_sma_source = input.source(close, title="-Slow SMA 2 Source")
slow_sma_length = input.int(10, title="-Slow SMA 2 Length-10-")

// Berechnung der SMAs
fast_sma = ta.sma(fast_sma_source, fast_sma_length)
slow_sma = ta.sma(slow_sma_source, slow_sma_length)

// Bedingungen für die Kreuzungen
bullish_cross = ta.crossover(fast_sma, slow_sma)
bearish_cross = ta.crossunder(fast_sma, slow_sma)

// U-Turn Buy/Sell Bedingungen
u_buy_signal = ta.crossover(close, ta.lowest(fast_sma, 5))
u_sell_signal = ta.crossunder(close, ta.highest(fast_sma, 5))

// Zusätzliche Bedingungen für Buy Signal 3 und Sell Signal 3
buy_signal_3 = ta.crossover(fast_sma, slow_sma) and ta.rising(slow_sma, 1)
sell_signal_3 = ta.crossunder(fast_sma, slow_sma) and ta.falling(slow_sma, 1)

// Zusätzliche Bedingungen für Buy Signal 4 und Sell Signal 4
buy_signal_4 = u_buy_signal and ta.rising(slow_sma, 1)
sell_signal_4 = u_sell_signal and ta.falling(slow_sma, 1)

// Bedingungen für Buy Signal 5 und Sell Signal 5
buy_signal_5 = u_buy_signal and ta.crossover(ta.lowest(slow_sma, 5), ta.lowest(slow_sma[1], 5))
sell_signal_5 = u_sell_signal and ta.crossunder(ta.highest(slow_sma, 5), ta.highest(slow_sma[1], 5))

// Bedingungen für Buy Signal 6 und Sell Signal 6
buy_signal_6 = ta.crossover(close, ta.lowest(fast_sma, 5)) and ta.rising(slow_sma, 5)
sell_signal_6 = ta.crossunder(close, ta.highest(fast_sma, 5)) and ta.falling(slow_sma, 5)

// Bedingungen für Buy Signal 7 und Sell Signal 7 (mit Muster)
pattern_buy = low > low[1] and low[1] > low[2] // Höhere Tiefs
pattern_sell = high < high[1] and high[1] < high[2] // Niedrigere Hochs

buy_signal_7 = bullish_cross and pattern_buy
sell_signal_7 = bearish_cross and pattern_sell

// Bedingungen für Buy Signal 8 und Sell Signal 8 (U-Turn und Muster)
uturn_buy = ta.crossover(fast_sma, fast_sma[1]) and low > low[1] and low[1] > low[2]
uturn_sell = ta.crossunder(fast_sma, fast_sma[1]) and high < high[1] and high[1] < high[2]

buy_signal_8 = uturn_buy
sell_signal_8 = uturn_sell

// Candlestick pattern functions
isBullish(o, c) => c > o
isBearish(o, c) => c < o
midpoint(price1, price2) => (price1 + price2) / 2
isDowntrend() => close[1] < close[2] and close[2] < close[3]
isUptrend() => close[1] > close[2] and close[2] > close[3]

// Bullish Patterns
bullishHammer = (low - math.min(open, close) > 2 * (math.max(open, close) - high)) and isDowntrend()
bullishInvertedHammer = (high - math.max(open, close) > 2 * (math.min(open, close) - low)) and isDowntrend()
bullishPiercing = (isBearish(close[1], open[1]) and isBullish(open, close) and open < close[1] and close > midpoint(open[1], close[1])) and isDowntrend()
bullishHarami = (isBearish(close[1], open[1]) and isBullish(open, close) and open > close[1] and close < open[1]) and isDowntrend()
bullishKicking = (isBearish(close[1], open[1]) and isBullish(open, close) and low > high[1]) and isDowntrend()
bullishEngulfing = (isBearish(close[1], open[1]) and isBullish(open, close) and open < close[1] and close > open[1]) and isDowntrend()
bullishMorningStar = (isBearish(close[2], open[2]) and (open[1] < close[2] and close[1] > open[2]) and isBullish(open, close) and close > midpoint(open[2], close[2])) and isDowntrend()
bullishBeltHold = (isBullish(open, close) and low == open and close > midpoint(high, low)) and isDowntrend()
bullishThreeWhiteSoldiers = (isBullish(open[2], close[2]) and isBullish(open[1], close[1]) and isBullish(open, close) and close[1] > close[2] and close > close[1]) and isDowntrend()

// Bearish Patterns
bearishHangingMan = (high - math.max(open, close) > 2 * (math.min(open, close) - low)) and isUptrend()
bearishShootingStar = (high - close > 2 * (open - low)) and isUptrend()
bearishBeltHold = (isBearish(open, close) and high == open and close < midpoint(high, low)) and isUptrend()
bearishHarami = (isBullish(close[1], open[1]) and isBearish(open, close) and open < close[1] and close > open[1]) and isUptrend()
bearishKicking = (isBullish(close[1], open[1]) and isBearish(open, close) and high < low[1]) and isUptrend()
bearishEngulfing = (isBullish(close[1], open[1]) and isBearish(open, close) and open > close[1] and close < open[1]) and isUptrend()
bearishEveningStar = (isBullish(close[2], open[2]) and (open[1] > close[2] and close[1] < open[2]) and isBearish(open, close) and close < midpoint(open[2], close[2])) and isUptrend()
bearishDarkCloud = (isBullish(close[1], open[1]) and isBearish(open, close) and open > close[1] and close < midpoint(open[1], close[1])) and isUptrend()
bearishThreeBlackCrows = (isBearish(open[2], close[2]) and isBearish(open[1], close[1]) and isBearish(open, close) and close[1] < close[2] and close < close[1]) and isUptrend()

// Buy and Sell Signal 9 (combined with candlestick patterns)
buy_signal_9 = u_buy_signal and (bullishHammer or bullishInvertedHammer or bullishPiercing or bullishHarami or bullishKicking or bullishEngulfing or bullishMorningStar or bullishBeltHold or bullishThreeWhiteSoldiers)
sell_signal_9 = u_sell_signal and (bearishHangingMan or bearishShootingStar or bearishBeltHold or bearishHarami or bearishKicking or bearishEngulfing or bearishEveningStar or bearishDarkCloud or bearishThreeBlackCrows)

// Buy und Sell Labels zeichnen
plotshape(series=bullish_cross, title="-Buy Signal-bullish_cross = ta.crossover(fast_sma, slow_sma)", location=location.belowbar, color=color.green, style=shape.labelup, text="BUY", textcolor=color.black, size=size.small)
plotshape(series=bearish_cross, title="-Sell Signal-bearish_cross = ta.crossunder(fast_sma, slow_sma)", location=location.abovebar, color=color.red, style=shape.labeldown, text="SELL", textcolor=color.black, size=size.small)

// U-Turn Buy und Sell Labels zeichnen
plotshape(series=u_buy_signal, title="-U-Turn Buy Signal-u_buy_signal = ta.crossover(close, ta.lowest(fast_sma, 5))", location=location.belowbar, color=color.green, style=shape.labelup, text="UT-Buy", textcolor=color.black, size=size.small)
plotshape(series=u_sell_signal, title="-U-Turn Sell Signal-u_sell_signal = ta.crossunder(close, ta.highest(fast_sma, 5))", location=location.abovebar, color=color.red, style=shape.labeldown, text="UT-Sell", textcolor=color.black, size=size.small)

// Zusätzliche Buy und Sell Labels zeichnen
plotshape(series=buy_signal_3, title="-Buy Signal 3-buy_signal_3 = ta.crossover(fast_sma, slow_sma) and ta.rising(slow_sma, 1)-", location=location.belowbar, color=color.green, style=shape.labelup, text="B3", textcolor=color.black, size=size.small)
plotshape(series=sell_signal_3, title="-Sell Signal 3-sell_signal_3 = ta.crossunder(fast_sma, slow_sma) and ta.falling(slow_sma, 1)-", location=location.abovebar, color=color.red, style=shape.labeldown, text="S3", textcolor=color.black, size=size.small)
plotshape(series=buy_signal_4, title="-Buy Signal 4-buy_signal_4 = u_buy_signal and ta.rising(slow_sma, 1)-", location=location.belowbar, color=color.green, style=shape.labelup, text="B4", textcolor=color.black, size=size.small)
plotshape(series=sell_signal_4, title="-Sell Signal 4-sell_signal_4 = u_sell_signal and ta.falling(slow_sma, 1)-", location=location.abovebar, color=color.red, style=shape.labeldown, text="S4", textcolor=color.black, size=size.small)

// Buy und Sell Labels für Signal 5 zeichnen
plotshape(series=buy_signal_5, title="-Buy Signal 5-buy_signal_5 = u_buy_signal and ta.crossover(ta.lowest(slow_sma, 5), ta.lowest(slow_sma[1], 5))-", location=location.belowbar, color=color.green, style=shape.labelup, text="B5", textcolor=color.black, size=size.small)
plotshape(series=sell_signal_5, title="-Sell Signal 5-sell_signal_5 = u_sell_signal and ta.crossunder(ta.highest(slow_sma, 5), ta.highest(slow_sma[1], 5))-", location=location.abovebar, color=color.red, style=shape.labeldown, text="S5", textcolor=color.black, size=size.small)

// Buy und Sell Labels für Signal 6 zeichnen
plotshape(series=buy_signal_6, title="-Buy Signal 6-buy_signal_6 = ta.crossover(close, ta.lowest(fast_sma, 5)) and ta.rising(slow_sma, 5)-", location=location.belowbar, color=color.green, style=shape.labelup, text="B6", textcolor=color.black, size=size.small)
plotshape(series=sell_signal_6, title="-Sell Signal 6-sell_signal_6 = ta.crossunder(close, ta.highest(fast_sma, 5)) and ta.falling(slow_sma, 5)-", location=location.abovebar, color=color.red, style=shape.labeldown, text="S6", textcolor=color.black, size=size.small)

// Buy und Sell Labels für Signal 7 zeichnen
plotshape(series=buy_signal_7, title="-Buy Signal 7-buy_signal_7 = bullish_cross and pattern_buy-pattern_buy = low > low[1] and low[1] > low[2] // Höhere Tiefs-", location=location.belowbar, color=color.green, style=shape.labelup, text="B7", textcolor=color.black, size=size.small)
plotshape(series=sell_signal_7, title="-Sell Signal 7-sell_signal_7 = bearish_cross and pattern_sell-pattern_sell = high < high[1] and high[1] < high[2] // Niedrigere Hochs-", location=location.abovebar, color=color.red, style=shape.labeldown, text="S7", textcolor=color.black, size=size.small)

// Buy und Sell Labels für Signal 8 zeichnen
plotshape(series=buy_signal_8, title="-Buy Signal 8-buy_signal_8 = uturn_buy-uturn_buy = ta.crossover(fast_sma, fast_sma[1]) and low > low[1] and low[1] > low[2]-", location=location.belowbar, color=color.green, style=shape.labelup, text="B8", textcolor=color.black, size=size.small)
plotshape(series=sell_signal_8, title="-Sell Signal 8-sell_signal_8 = uturn_sell-uturn_sell = ta.crossunder(fast_sma, fast_sma[1]) and high < high[1] and high[1] < high[2]-", location=location.abovebar, color=color.red, style=shape.labeldown, text="S8", textcolor=color.black, size=size.small)

// Buy und Sell Labels für Signal 9 zeichnen
plotshape(series=buy_signal_9, title="-Buy Signal 9-buy_signal_9 = u_buy_signal and (bullishHammer or bullishInvertedHammer or bullishPiercing or bullishHarami or bullishKicking or bullishEngulfing or bullishMorningStar or bullishBeltHold or bullishThreeWhiteSoldiers)-", location=location.belowbar, color=color.green, style=shape.labelup, text="B9", textcolor=color.black, size=size.small)
plotshape(series=sell_signal_9, title="-Sell Signal 9-sell_signal_9 = u_sell_signal and (bearishHangingMan or bearishShootingStar or bearishBeltHold or bearishHarami or bearishKicking or bearishEngulfing or bearishEveningStar or bearishDarkCloud or bearishThreeBlackCrows)-", location=location.abovebar, color=color.red, style=shape.labeldown, text="S9", textcolor=color.black, size=size.small)

// SMAs auf dem Chart anzeigen
plot(fast_sma, title="Fast SMA 5", color=color.yellow, linewidth=2)
plot(slow_sma, title="Slow SMA 10", color=color.green, linewidth=2)
