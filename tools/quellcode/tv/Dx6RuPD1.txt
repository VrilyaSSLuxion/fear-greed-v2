//@version=5
indicator("-VrilyaSS-MA-Crossover-50-SMA+200-SMA-", overlay=true)

// Inputs
fastLength = input(50, title="-Fast-MA-Length-50-")
slowLength = input(200, title="-Slow-MA-Length-200-")
src = input(close, title="-Source-close-")

// Calculate moving averages
fastMA = ta.sma(src, fastLength)
slowMA = ta.sma(src, slowLength)

// Plot moving averages with custom colors and thicker lines (linewidth=2)
plot(fastMA, color=color.white, title="-Fast-MA-", linewidth=2)
plot(slowMA, color=#fccbcd, title="-Slow-MA-", linewidth=3)

// Generate trading signals
longCondition = ta.crossover(fastMA, slowMA)
shortCondition = ta.crossunder(fastMA, slowMA)

// Plot arrows for buy/sell signals
plotshape(series=longCondition, title="-Buy-Signal-", location=location.belowbar, color=color.green, textcolor=color.black, style=shape.labelup, text="Buy", size=size.tiny)
plotshape(series=shortCondition, title="-Sell-Signal-", location=location.abovebar, color=color.red, textcolor=color.black, style=shape.labeldown, text="Sell", size=size.tiny)

// Alert conditions
alertcondition(longCondition, title="-Buy-Signal-VrilyaSS-MA-Crossover-50-SMA+200-SMA-", message="-Buy-Fast-MA-50-crossed above Slow-MA-200-")
alertcondition(shortCondition, title="-Sell-Signal-VrilyaSS-MA-Crossover-50-SMA+200-SMA-", message="-Sell-Fast-MA-50-crossed below Slow-MA-200-")

// Chart Title
var label titleLabel = na
if not na(titleLabel)
    label.delete(titleLabel)

showTitle = input(false, title="Show Chart Title")
if showTitle
    titleLabel := label.new(x=bar_index, y=high, text="-Moving-Average-Crossover-Indicator-", style=label.style_label_down, color=color.blue, textcolor=color.white, size=size.tiny)
