oscRaw(simple int length, simple bool wait) =>
    float hh = ta.highest(high, length)
    float ll = ta.lowest(low, length)
    float d  = hh - ll
    float v  = d > 0 ? (close - ll) / d * 100 - 100 : -50.0
    wait ? v[1] : v

calcOscillator(simple string tf, simple int len, simple bool wait) =>
    request.security(syminfo.tickerid, tf, oscRaw(len, wait), lookahead=barmerge.lookahead_off)

detectSignals(float osc, float upper, float lower) =>
    var int   nextSig = 0
    var float seqTop  = -105.0
    var float seqBot  = 5.0
    var float peakTop = na
    var float peakBot = na
    var bool  inTop   = false
    var bool  inBot   = false
    bool trigTop = false
    bool trigBot = false

    if not na(osc)
        if osc > upper
            if not inTop
                inTop   := true
                peakTop := osc
            else if osc > peakTop
                peakTop := osc
        else if inTop
            if not na(peakTop) and peakTop > upper
                if nextSig == 0 or nextSig == 1 or (nextSig == -1 and peakTop > seqTop)
                    trigTop := true
                    seqTop  := peakTop
                    nextSig := -1
            inTop   := false
            peakTop := na

        if osc < lower
            if not inBot
                inBot   := true
                peakBot := osc
            else if osc < peakBot
                peakBot := osc
        else if inBot
            if not na(peakBot) and peakBot < lower
                if nextSig == 0 or nextSig == -1 or (nextSig == 1 and peakBot < seqBot)
                    trigBot := true
                    seqBot  := peakBot
                    nextSig := 1
            inBot   := false
            peakBot := na
    [trigTop, trigBot]

baseLong  = ((armLongEvt  and armOnSignal and sigLong)  or showAllFractals) and isFractalLow
baseShort = ((armShortEvt and armOnSignal and sigShort) or showAllFractals) and isFractalHigh

// Erweitert: erstes Fraktal nach dem Auslöser + jedes tiefere/höhere danach
betterLong  = markEveryFract or na(armRef) or low[1]  < armRef
betterShort = markEveryFract or na(armRef) or high[1] > armRef

followLong  = rescanMode and armDir ==  1 and isFractalLow  and betterLong
followShort = rescanMode and armDir == -1 and isFractalHigh and betterShort

makeLong  = baseLong  or followLong
makeShort = baseShort or followShort

// Referenz fortschreiben
if makeLong
    armRef := na(armRef) ? low[1]  : math.min(armRef, low[1])
    armDir := 1
if makeShort
    armRef := na(armRef) ? high[1] : math.max(armRef, high[1])
    armDir := -1

isFollow = (followLong or followShort) and not (baseLong or baseShort)

//============================================
// 15. OSZILLATOR-FENSTER
//============================================
plot(showInstance1 ? osc1 : na, "Instanz1", color1, linewidth=width1)
plot(showInstance2 ? osc2 : na, "Instanz2", color2, linewidth=width2)
plot(showInstance3 ? osc3 : na, "Instanz3", color3, linewidth=width3)

hline(upperLine, "Obere Zone", color=showZones ? upperColor : color.new(color.gray, 100), linestyle=upperStyleIn == "DASHED" ? hline.style_dashed : upperStyleIn == "DOTTED" ? hline.style_dotted : hline.style_solid, linewidth=upperWidth)
hline(lowerLine, "Untere Zone", color=showZones ? lowerColor : color.new(color.gray, 100), linestyle=lowerStyleIn == "DASHED" ? hline.style_dashed : lowerStyleIn == "DOTTED" ? hline.style_dotted : hline.style_solid, linewidth=lowerWidth)
hline(0,    "Mitte", color.gray, hline.style_dashed)
hline(-100, "Unten", color.gray, hline.style_dashed)

if showExtremArrow and anyExShort
    label.new(bar_index, high, extremText, color=color.new(extremBgColor, extremBgTransp), style=toLabelStyle(extremStyleDn), size=toSize(extremSize), textcolor=extremColor, force_overlay=true)

if drawArrow and makeShort
    label.new(bar_index[1], high[1], shortText, color=color.new(shortBgColor, shortBgTransp), style=toLabelStyle(shortStyle), size=toSize(shortSize), textcolor=(isFollow and rescanTint) ? rescanColS : shortEntryColor, force_overlay=true)

//============================================
// 17. LINIEN-VERWALTUNG (Fraktal + TP/SL Level)
//============================================
var line[]  allLines  = array.new_line()
var label[] allLabels = array.new_label()
var int[]   setSizes  = array.new_int()
var int[]   setLbls   = array.new_int()
var int     fractalCount = 0

string lineStyleStr = toLineStyle(lineStyleIn)
bool drawLines = (not isFollow or rescanLines)

if (makeLong or makeShort) and drawLines and (showFractalLines or tpMaster)
    fractalCount += 1

    int   anchor = bar_index[1]
    float fHigh  = high[1]
    float fLow   = low[1]
    float risk   = fHigh - fLow
    float entry  = makeLong ? fHigh : fLow
    int   dir    = makeLong ? 1 : -1

    // --- "Bis nächstes Fraktal": vorheriges Set einfrieren ---
    if extendMode == "Bis nächstes Fraktal" and array.size(setSizes) > 0
        int lastL = array.get(setSizes, array.size(setSizes) - 1)
        int lastB = array.get(setLbls,  array.size(setLbls)  - 1)
        int totL  = array.size(allLines)
        int totB  = array.size(allLabels)
        if lastL > 0
            for k = totL - lastL to totL - 1
                line.set_x2(array.get(allLines, k), anchor)
        if lastB > 0
            for k = totB - lastB to totB - 1
                label.set_x(array.get(allLabels, k), anchor)

    int endBar = switch extendMode
        "Feste Bars"    => anchor + extendBars
        "Live + X Bars" => bar_index + extendBars
        => bar_index

    int drawnL = 0
    int drawnB = 0

    // --- Entry- und SL-Linie ---
    if showFractalLines
        color hiC = makeLong ? longHighColor : shortHighColor
        color loC = makeLong ? longLowColor  : shortLowColor
        array.push(allLines, line.new(anchor, fHigh, endBar, fHigh, color=hiC, width=lineWidth, style=lineStyleStr, force_overlay=true))
        array.push(allLines, line.new(anchor, fLow,  endBar, fLow,  color=loC, width=lineWidth, style=lineStyleStr, force_overlay=true))
        drawnL += 2
        if tpShowLabels
            array.push(allLabels, label.new(endBar, fHigh, makeLong ? "Entry" : "SL", color=color.new(color.black, 100), textcolor=hiC, style=label.style_label_left, size=toSize(tpLabelSize), force_overlay=true))
            array.push(allLabels, label.new(endBar, fLow,  makeLong ? "SL" : "Entry", color=color.new(color.black, 100), textcolor=loC, style=label.style_label_left, size=toSize(tpLabelSize), force_overlay=true))
            drawnB += 2

    // --- TP/SL Level-Linien ---
    if tpMaster and risk > 0
        for i = 0 to array.size(lvMult) - 1
            if array.get(lvOn, i)
                float lvl = entry + dir * array.get(lvMult, i) * risk
                color lc  = array.get(lvCol, i)
                array.push(allLines, line.new(anchor, lvl, endBar, lvl, color=lc, width=array.get(lvWid, i), style=toLineStyle(array.get(lvSty, i)), force_overlay=true))
                drawnL += 1
                if tpShowLabels
                    array.push(allLabels, label.new(endBar, lvl, array.get(lvName, i), color=color.new(color.black, 100), textcolor=lc, style=label.style_label_left, size=toSize(tpLabelSize), force_overlay=true))
                    drawnB += 1

    // --- Set registrieren & alte Sets löschen ---
    if drawnL > 0
        array.push(setSizes, drawnL)
        array.push(setLbls,  drawnB)
        int maxSets = fractalLineCount > 0 ? fractalLineCount : 100
        while array.size(setSizes) > maxSets
            int oldL = array.shift(setSizes)
            int oldB = array.shift(setLbls)
            for k = 1 to oldL
                if array.size(allLines) > 0
                    line.delete(array.shift(allLines))
            if oldB > 0
                for k = 1 to oldB
                    if array.size(allLabels) > 0
                        label.delete(array.shift(allLabels))

// --- Live-Nachführung nur auf der letzten Bar ---
if barstate.islast
    if extendMode == "Live" or extendMode == "Live + X Bars"
        int tgt = extendMode == "Live" ? bar_index : bar_index + extendBars
        if array.size(allLines) > 0
            for i = 0 to array.size(allLines) - 1
                line.set_x2(array.get(allLines, i), tgt)
        if array.size(allLabels) > 0
            for i = 0 to array.size(allLabels) - 1
                label.set_x(array.get(allLabels, i), tgt)
    else if extendMode == "Bis nächstes Fraktal" and array.size(setSizes) > 0
        int lastL = array.get(setSizes, array.size(setSizes) - 1)
        int lastB = array.get(setLbls,  array.size(setLbls)  - 1)
        int totL  = array.size(allLines)
        int totB  = array.size(allLabels)
        if lastL > 0
            for i = totL - lastL to totL - 1
                line.set_x2(array.get(allLines, i), bar_index)
        if lastB > 0
            for i = totB - lastB to totB - 1
                label.set_x(array.get(allLabels, i), bar_index)
