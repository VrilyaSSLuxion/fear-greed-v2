//@version=5
indicator(title="-Quick Trade Indicator-(QTI)-(TV-ALLIGATOR-(LIP-TEETH)+RSI-", shorttitle="-Quick Trade Indicator-(QTI)-(TV-ALLIGATOR-(LIP-TEETH)+RSI-", overlay=true)

// Williams Alligator Indicator
smma(src, length) =>
    smma =  0.0
    smma := na(smma[1]) ? ta.sma(src, length) : (smma[1] * (length - 1) + src) / length

jawLength = input.int(13, minval=1, title="-Jaw-Length-13-")
teethLength = input.int(8, minval=1, title="-Teeth-Length-8-")
lipsLength = input.int(5, minval=1, title="-Lips-Length-5-")
jawOffset = input.int(8, title="-Jaw-Offset-8-")
teethOffset = input.int(5, title="-Teeth-Offset-5-")
lipsOffset = input.int(3, title="-Lips-Offset-3-")

jaw = smma(hl2, jawLength)
teeth = smma(hl2, teethLength)
lips = smma(hl2, lipsLength)

plot(jaw, "-Jaw-", offset=jawOffset, color=color.blue, linewidth=2)
plot(teeth, "-Teeth-", offset=teethOffset, color=color.red, linewidth=2)
plot(lips, "-Lips-", offset=lipsOffset, color=color.green, linewidth=2)

// RSI settings
rsi_source = input.source(close, title="-RSI-Source-close-", inline="RSI")
rsi_period = input.int(14, title="-RSI-Period-14-", inline="RSI")
rsi_buy_level = input.int(53, title="-RSI-Buy-Level-30-", inline="RSI")
rsi_sell_level = input.int(47, title="-RSI-Sell-Level-70-", inline="RSI")

// Calculate RSI
rsi = ta.rsi(rsi_source, rsi_period)

// Buy and Sell signals based on Alligator and RSI
bullish_crossover = ta.crossover(lips, teeth) and rsi < rsi_buy_level
bearish_crossover = ta.crossunder(lips, teeth) and rsi > rsi_sell_level

// Plot Buy and Sell signals
plotshape(series=bullish_crossover, title="-Buy Signal-", location=location.belowbar, color=color.green, textcolor=color.black, style=shape.labelup, text="BUY", size=size.tiny)
plotshape(series=bearish_crossover, title="-Sell Signal-", location=location.abovebar, color=color.red, textcolor=color.black, style=shape.labeldown, text="SELL", size=size.tiny)

// Plot RSI for reference
plot(rsi, title="-RSI-", color=color.yellow, linewidth=2, display = display.none)
hline(rsi_buy_level, title="-RSI-Buy-Level-", color=color.green, linewidth=3, linestyle=hline.style_solid, display = display.none)
hline(rsi_sell_level, title="-RSI-Sell-Level-", color=color.red, linewidth=3, linestyle=hline.style_solid, display = display.none)
