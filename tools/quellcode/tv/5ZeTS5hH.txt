//@version=5
indicator(title="-VrilyaSS-TSA TDI-TraderJe-Version#1-Condition+Divergences-0607-", shorttitle="-VrilyaSS-TSA TDI-TraderJe-Version#1-Condition+Divergences-0607-", format=format.price, precision=2)

// TDI calculation
rsi = ta.rsi(close, 8)
ma = ta.sma(rsi, 34)
offs = 1.6185 * ta.stdev(rsi, 34)
up = ma + offs
dn = ma - offs
mid = (up + dn) / 2
fastMA = ta.sma(rsi, 2)

// Plotting bands and lines
hline(79, "-OverBought-default-Color-gray-5-100%-Line Width-1-dashed Line-75-", color=color.gray, linestyle=hline.style_dashed, editable=true)
hline(21, "-OverSold-default-Color-gray-5-100%-Line Width-1-dashed Line-25-", color=color.gray, linestyle=hline.style_dashed, editable=true)
hline(89, "-Upper-Horizontal Line-default-Color-gray-5-100%-Line Width-1-dashed Line-80-", color=color.gray, linestyle=hline.style_dashed, editable=true)
hline(50, "-Middle-Horizontal Line-default-Color-gray-5-100%-Line Width-1-dashed Line-50-", color=color.white, linestyle=hline.style_dashed, linewidth=2, editable=true)
hline(11, "-Lower-Horizontal Line-default-Color-gray-5-100%-Line Width-1-dashed Line-20-", color=color.gray, linestyle=hline.style_dashed, editable=true)

upl = plot(up, "-Upper Band-", color=color.new(color.red, 0), linewidth=3)
dnl = plot(dn, "-Lower Band-", color=color.new(color.green, 0), linewidth=3)
midl = plot(mid, "-Middle of Bands-", color=color.new(color.white, 0), linewidth=2)
plot(fastMA, "-Fast MA-", color=color.new(color.yellow, 0), linewidth=2)

// Checkboxen für Kauf- und Verkaufssignale mit Beschreibungen
activate_buy_signal1 = input(true, title="Activate Buy Signal 1 = fastMA <= 30 and fastMA < mid and fastMA < dn")
activate_sell_signal1 = input(true, title="Activate Sell Signal 1 = fastMA >= 70 and fastMA > mid and fastMA > up")

activate_buy_signal2 = input(true, title="Activate Buy Signal 2 = fastMA <= 20 and fastMA < mid and fastMA < dn")
activate_sell_signal2 = input(true, title="Activate Sell Signal 2 = fastMA >= 80 and fastMA > mid and fastMA > up")

activate_buy_signal3 = input(true, title="Activate Buy Signal 3 = fastMA crosses above mid")
activate_sell_signal3 = input(true, title="Activate Sell Signal 3 = fastMA crosses below mid")

activate_buy_signal4 = input(true, title="Activate Buy Signal 4 = fastMA crosses above dn")
activate_sell_signal4 = input(true, title="Activate Sell Signal 4 = fastMA crosses below up")

// Buy and sell conditions based on crossover with mid
buyCondition1 = activate_buy_signal1 and (fastMA <= 30 and fastMA < mid and fastMA < dn)
sellCondition1 = activate_sell_signal1 and (fastMA >= 70 and fastMA > mid and fastMA > up)

buyCondition2 = activate_buy_signal2 and (fastMA <= 20 and fastMA < mid and fastMA < dn)
sellCondition2 = activate_sell_signal2 and (fastMA >= 80 and fastMA > mid and fastMA > up)

buyCondition3 = activate_buy_signal3 and ta.crossover(fastMA, mid)
sellCondition3 = activate_sell_signal3 and ta.crossunder(fastMA, mid)

buyCondition4 = activate_buy_signal4 and ta.crossover(fastMA, dn)
sellCondition4 = activate_sell_signal4 and ta.crossunder(fastMA, up)

// Signal labels
if (buyCondition1)
    label.new(bar_index, dn, text="BUY 1", style=label.style_label_up, color=color.green, textcolor=color.black, yloc=yloc.price, size=size.small)
if (sellCondition1)
    label.new(bar_index, up, text="SELL 1", style=label.style_label_down, color=color.red, textcolor=color.black, yloc=yloc.price, size=size.small)

if (buyCondition2)
    label.new(bar_index, dn, text="BUY 2", style=label.style_label_up, color=color.blue, textcolor=color.black, yloc=yloc.price, size=size.small)
if (sellCondition2)
    label.new(bar_index, up, text="SELL 2", style=label.style_label_down, color=color.orange, textcolor=color.black, yloc=yloc.price, size=size.small)

if (buyCondition3)
    label.new(bar_index, mid, text="BUY 3", style=label.style_label_up, color=color.green, textcolor=color.black, yloc=yloc.price, size=size.small)
if (sellCondition3)
    label.new(bar_index, mid, text="SELL 3", style=label.style_label_down, color=color.red, textcolor=color.black, yloc=yloc.price, size=size.small)

if (buyCondition4)
    label.new(bar_index, dn, text="BUY 4", style=label.style_label_up, color=color.green, textcolor=color.black, yloc=yloc.price, size=size.small)
if (sellCondition4)
    label.new(bar_index, up, text="SELL 4", style=label.style_label_down, color=color.red, textcolor=color.black, yloc=yloc.price, size=size.small)

// Alerts for buy and sell signals
alertcondition(buyCondition1, title="Buy Alert 1", message="Buy Signal 1 on {{ticker}}")
alertcondition(sellCondition1, title="Sell Alert 1", message="Sell Signal 1 on {{ticker}}")

alertcondition(buyCondition2, title="Buy Alert 2", message="Buy Signal 2 on {{ticker}}")
alertcondition(sellCondition2, title="Sell Alert 2", message="Sell Signal 2 on {{ticker}}")

alertcondition(buyCondition3, title="Buy Alert 3", message="Buy Signal 3 on {{ticker}}")
alertcondition(sellCondition3, title="Sell Alert 3", message="Sell Signal 3 on {{ticker}}")

alertcondition(buyCondition4, title="Buy Alert 4", message="Buy Signal 4 on {{ticker}}")
alertcondition(sellCondition4, title="Sell Alert 4", message="Sell Signal 4 on {{ticker}}")

// Alarmbedingungen
finUp = fastMA > 75 and fastMA > up
finDn = fastMA < 25 and fastMA < dn
alert = finUp or finDn

if alert
    alert('{"content":"' + syminfo.ticker + ' ' + timeframe.period + ' Sharkfin"}', alert.freq_once_per_bar_close)

// Divergenz-Erkennungslogik
lbR = input.int(title="-Pivot Lookback Right-5-", defval=5, display=display.data_window)
lbL = input.int(title="-Pivot Lookback Left-5-", defval=5, display=display.data_window)
rangeUpper = input.int(title="-Max of Lookback Range-60-", defval=60, display=display.data_window)
rangeLower = input.int(title="-Min of Lookback Range-5-", defval=5, display=display.data_window)
plotBull = input.bool(title="-Plot Bullish-true-", defval=true, display=display.data_window)
plotHiddenBull = input.bool(title="-Plot Hidden-Bullish-true-", defval=true, display=display.data_window)
plotBear = input.bool(title="-Plot Bearish-true-", defval=true, display=display.data_window)
plotHiddenBear = input.bool(title="-Plot Hidden-Bearish-true-", defval=true, display=display.data_window)

bullColor = input.color(title="-Regular-Bullish-Label-Color-", defval=color.green, group="Regular Bullish", inline="color")
bullTextColor = input.color(title="-Regular-Bullish-Label-Text-Color-", defval=color.white, group="Regular Bullish", inline="textColor")

hiddenBullColor = input.color(title="-Hidden-Bullish-Label-Color-", defval=color.new(color.green, 50), group="Hidden Bullish", inline="color")
hiddenBullTextColor = input.color(title="-Hidden-Bullish-Label-Text-Color-", defval=color.white, group="Hidden Bullish", inline="textColor")

bearColor = input.color(title="-Regular-Bearish-Label-Color-", defval=color.red, group="Regular Bearish", inline="color")
bearTextColor = input.color(title="-Regular-Bearish-Label-Text-Color-", defval=color.white, group="Regular Bearish", inline="textColor")

hiddenBearColor = input.color(title="-Hidden-Bearish-Label-Color-", defval=color.new(color.red, 50), group="Hidden Bearish", inline="color")
hiddenBearTextColor = input.color(title="-Hidden-Bearish-Label-Text-Color-", defval=color.white, group="Hidden Bearish", inline="textColor")

noneColor = color.new(color.white, 100)

// Pivots erkennen
plFound = na(ta.pivotlow(fastMA, lbL, lbR)) ? false : true
phFound = na(ta.pivothigh(fastMA, lbL, lbR)) ? false : true
_inRange(cond) =>
    bars = ta.barssince(cond == true)
    rangeLower <= bars and bars <= rangeUpper

// Reguläre Bullish-Divergenz
oscHL = fastMA[lbR] > ta.valuewhen(plFound, fastMA[lbR], 1) and _inRange(plFound[1])
priceLL = low[lbR] < ta.valuewhen(plFound, low[lbR], 1)
bullCondAlert = priceLL and oscHL and plFound
bullCond = plotBull and bullCondAlert

plot(plFound ? fastMA[lbR] : na, offset=-lbR, title="-Regular Bullish-", linewidth=2, color=(bullCond ? bullColor : noneColor), display=display.pane)

plotshape(bullCond ? fastMA[lbR] : na, offset=-lbR, title="-Regular Bullish-Label-", text=" Bull ", style=shape.labelup, location=location.absolute, color=bullColor, textcolor=bullTextColor)

// Versteckte Bullish-Divergenz
oscLL = fastMA[lbR] < ta.valuewhen(plFound, fastMA[lbR], 1) and _inRange(plFound[1])
priceHL = low[lbR] > ta.valuewhen(plFound, low[lbR], 1)
hiddenBullCondAlert = priceHL and oscLL and plFound
hiddenBullCond = plotHiddenBull and hiddenBullCondAlert

plot(plFound ? fastMA[lbR] : na, offset=-lbR, title="-Hidden Bullish-", linewidth=2, color=(hiddenBullCond ? hiddenBullColor : noneColor), display=display.pane)

plotshape(hiddenBullCond ? fastMA[lbR] : na, offset=-lbR, title="-Hidden Bullish-Label-", text=" H Bull ", style=shape.labelup, location=location.absolute, color=hiddenBullColor, textcolor=hiddenBullTextColor)

// Reguläre Bearish-Divergenz
oscLH = fastMA[lbR] < ta.valuewhen(phFound, fastMA[lbR], 1) and _inRange(phFound[1])
priceHH = high[lbR] > ta.valuewhen(phFound, high[lbR], 1)
bearCondAlert = priceHH and oscLH and phFound
bearCond = plotBear and bearCondAlert

plot(phFound ? fastMA[lbR] : na, offset=-lbR, title="-Regular Bearish-", linewidth=2, color=(bearCond ? bearColor : noneColor), display=display.pane)

plotshape(bearCond ? fastMA[lbR] : na, offset=-lbR, title="-Regular Bearish-Label-", text=" Bear ", style=shape.labeldown, location=location.absolute, color=bearColor, textcolor=bearTextColor)

// Versteckte Bearish-Divergenz
oscHH = fastMA[lbR] > ta.valuewhen(phFound, fastMA[lbR], 1) and _inRange(phFound[1])
priceLH = high[lbR] < ta.valuewhen(phFound, high[lbR], 1)
hiddenBearCondAlert = priceLH and oscHH and phFound
hiddenBearCond = plotHiddenBear and hiddenBearCondAlert

plot(phFound ? fastMA[lbR] : na, offset=-lbR, title="-Hidden Bearish-", linewidth=2, color=(hiddenBearCond ? hiddenBearColor : noneColor), display=display.pane)

plotshape(hiddenBearCond ? fastMA[lbR] : na, offset=-lbR, title="-Hidden Bearish-Label-", text=" H Bear ", style=shape.labeldown, location=location.absolute, color=hiddenBearColor, textcolor=hiddenBearTextColor)

// Alerts for divergences
alertcondition(bullCondAlert, title="Regular Bullish Divergence", message="Found a new Regular Bullish Divergence, Pivot Lookback Right number of bars to the left of the current bar")
alertcondition(hiddenBullCondAlert, title="Hidden Bullish Divergence", message="Found a new Hidden Bullish Divergence, Pivot Lookback Right number of bars to the left of the current bar")
alertcondition(bearCondAlert, title="Regular Bearish Divergence", message="Found a new Regular Bearish Divergence, Pivot Lookback Right number of bars to the left of the current bar")
alertcondition(hiddenBearCondAlert, title="Hidden Bearish Divergence", message="Found a new Hidden Bearish Divergence, Pivot Lookback Right number of bars to the left of the current bar")
