//@version=5
indicator("Volume Spike Momentum", overlay=true)

// Input settings
volumeSpikeMultiplier = input.float(2.0, title="Volume Spike Multiplier")
lookbackPeriod = input.int(20, title="Volume Lookback Period")
momentumThreshold = input.float(1.5, title="Price Momentum Threshold")
rsiPeriod = input.int(14, title="RSI Period")
rsiOverbought = input.int(70, title="RSI Overbought Level")
rsiOversold = input.int(30, title="RSI Oversold Level")
enableBuySignal = input.bool(true, title="Enable Buy Signal")
enableSellSignal = input.bool(true, title="Enable Sell Signal")

// Calculate the average volume over the lookback period
avgVolume = ta.sma(volume, lookbackPeriod)

// Detect Volume Spikes
volumeSpike = volume > avgVolume * volumeSpikeMultiplier

// Calculate Momentum (Rate of Change)
priceMomentum = ta.roc(close, lookbackPeriod)

// Calculate RSI for additional filtering
rsi = ta.rsi(close, rsiPeriod)

// Buy and Sell Conditions
buyCondition = enableBuySignal and volumeSpike and priceMomentum > momentumThreshold and rsi < rsiOverbought
sellCondition = enableSellSignal and volumeSpike and priceMomentum < -momentumThreshold and rsi > rsiOversold

// Plot Buy and Sell Signals
plotshape(series=buyCondition, title="Buy Signal", location=location.belowbar, color=color.green, style=shape.labelup, text="BUY")
plotshape(series=sellCondition, title="Sell Signal", location=location.abovebar, color=color.red, style=shape.labeldown, text="SELL")

// Alerts
alertcondition(buyCondition, title="Buy Signal", message="Buy Signal detected")
alertcondition(sellCondition, title="Sell Signal", message="Sell Signal detected")
