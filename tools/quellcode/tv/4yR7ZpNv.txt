//@version=5
indicator(title="Quick Trade Indicator (QTI9)", shorttitle="QTI9", overlay=false)

// Inputs
length = 9
src = close

// Custom function for Weighted Moving Average (WMA)
wma(src, length) =>
    sum = 0.0
    sumw = 0.0
    for i = 0 to length - 1
        weight = (length * (length + 1)) / 2 - (i * (i + 1)) / 2
        sum := sum + src[i] * weight
        sumw := sumw + weight
    sum / sumw

// Calculate QTI9 using WMA
qti9 = wma(src - wma(src, length), length)

// Determine trend direction
trendUp = qti9 > qti9[1]
trendDown = qti9 < qti9[1]

// Plot QTI9
plot(qti9, color=color.blue, title="QTI9")

// Plot signals
plotshape(trendUp, title="Buy Signal", location=location.belowbar, color=color.green, style=shape.labelup, text="Buy")
plotshape(trendDown, title="Sell Signal", location=location.abovebar, color=color.red, style=shape.labeldown, text="Sell")

// Alerts
alertcondition(trendUp, title="Buy Signal", message="QTI9 Buy Signal")
alertcondition(trendDown, title="Sell Signal", message="QTI9 Sell Signal")
