//@version=5
indicator("-VrilyaSS-True Williams Alligator (SMMA)-with Buy+Sell-Signals-", overlay=true)

// SMMA function definition
smma(src, length) =>
    var float smma = na
    smma := na(smma) ? ta.sma(src, length) : (smma[1] * (length - 1) + src) / length
    smma

// Input settings for Alligator
jawLength = input.int(13, "-Jaw-Length-13-")
teethLength = input.int(8, "-Teeth-Length-8-")
lipsLength = input.int(5, "-Lips-Length-5-")
jawOffset = input.int(8, "-Jaw-Offset-8-")
teethOffset = input.int(5, "-Teeth-Offset-5-")
lipsOffset = input.int(3, "-Lips-Offset-3-")

// Calculate Alligator lines
jaw = smma(hl2, jawLength)
teeth = smma(hl2, teethLength)
lips = smma(hl2, lipsLength)

// Plot Alligator lines
plot(jaw, "-Jaw-red-11-Line-4-", color.red, linewidth = 4, offset=jawOffset)
plot(teeth, "-Teeth-purple-19-Line-3-", color = #9c27b0, linewidth = 3, offset=teethOffset)
plot(lips, "-Lips-aqua-16-Line-2-", color.aqua, linewidth = 2, offset=lipsOffset)

// Buy and Sell conditions
buyCondition = ta.crossover(jaw, teeth) and jaw > lips
sellCondition = ta.crossunder(jaw, teeth) and jaw < lips

// Plot Buy and Sell Signals using plotshape
plotshape(series=buyCondition, title="-Buy-Signal-", style=shape.labelup, location=location.belowbar, color=color.green, text="Buy", textcolor=color.black, size=size.tiny)
plotshape(series=sellCondition, title="-Sell-Signal-", style=shape.labeldown, location=location.abovebar, color=color.red, text="Sell", textcolor=color.black, size=size.tiny)
