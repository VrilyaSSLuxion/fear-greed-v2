//@version=5
strategy("RSI & Stochastic Strategy", overlay=true)

// RSI Settings
rsiLength = input.int(14, "RSI Length")
rsiOversold = input.int(30, "RSI Oversold Level")
rsiOverbought = input.int(70, "RSI Overbought Level")

// Stochastic Settings
stochLength = input.int(14, "Stochastic Length")
smoothK = input.int(3, "Smoothing for %K")
smoothD = input.int(3, "Smoothing for %D")
overbought = input.int(80, "Overbought Level")
oversold = input.int(20, "Oversold Level")

// RSI Calculation
rsiValue = ta.rsi(close, rsiLength)

// Stochastic Calculation
k = ta.sma(ta.stoch(close, high, low, stochLength), smoothK)
d = ta.sma(k, smoothD)

// Buy Signal
buySignal = (rsiValue < rsiOversold) and (k < oversold) and ta.crossover(k, d)

// Sell Signal
sellSignal = (rsiValue > rsiOverbought) and (k > overbought) and ta.crossunder(k, d)

// Plot Buy and Sell Signals
plotshape(series=buySignal, title="Buy Signal", location=location.belowbar, color=color.green, style=shape.triangleup, size=size.small)
plotshape(series=sellSignal, title="Sell Signal", location=location.abovebar, color=color.red, style=shape.triangledown, size=size.small)

// Execute Buy and Sell Orders
if (buySignal)
    strategy.entry("Buy", strategy.long)
if (sellSignal)
    strategy.close("Buy")
