// @type                A custom type for storing price-time coordinates.
// @field y             A value on the y-axis (price).
// @field x             A value on the x-axis (time).
type point
    float y
    int   x


// @type                A custom type for representing and displaying pivots for Zig Zag construction.
// @field start         References a `point` object containing the coordinates of the previous pivot point.
// @field end           References a `point` object containing the coordinates of the current pivot point.
// @field dir           Indicates the direction of movement for the pivot: 1 for upward, and -1 for downward.
// @field ln            References a `line` object to draw a line connecting the current and previous pivot points.
// @field bars          The number of bars in the pivot. The default is 0.
type pivotPoint
    point start
    point end
    int   dir
    line  ln
    int   bars = 0


// @type                A custom type for displaying Fibonacci retracement levels.
// @field level         The Fibonacci level's ratio (e.g., 0.618).
// @field fibColor      The color of the level's lines and label.
// @field fibLine       References a `line` object to draw a horizontal line across the drawing range.
// @field extLine       References a `line` object to draw another horizontal line up to the last available bar.
// @field fibLabel      References a `label` object to display level information next to the horizontal lines.
type fibLevel
    float level
    color fibColor
    line  fibLine
    line  extLine
    label fibLabel

// @function            Converts a value from exponential to logarithmic form.
// @param value         (series float) The value to convert to log.
// @returns             (float) The log value.
toLog(series float value) =>
    float exponent = math.abs(value)
    float product  = math.log10(exponent + 0.0001) + 4
    float result   = exponent < 1e-8 ? 0 : value < 0 ? - product : product


// @function            Converts a value from logarithmic to exponential form.
// @param value         (series float) The value to convert from log.
// @returns             (float) The exponential value.
fromLog(series float value) =>
    float exponent = math.abs(value)
    float product  = math.pow(10, exponent - 4) - 0.0001
    float result   = exponent < 1e-8 ? 0 : value < 0 ? - product : product


// @function            Calculates a Fibonacci price level based on high and low prices, the retracement direction.
//                      The function calculates log-scaled levels if the "Log scale Fibs" input is enabled.
// @param value         (series float) The Fibonacci level's ratio (e.g., 0.618).
// @param hiPrice       (series float) The retracement high price.
// @param loPrice       (series float) The retracement low price.
// @param isBull        (series bool) If `true`, the retracement direction is bullish. If `false`, it is bearish.
// @returns             (float) The price value of the Fibonacci level.
getPrice(series float value, series float hiPrice, series float loPrice, series bool isBull) =>
    float hiPoint  = useLogScaleInput ? toLog(hiPrice) : hiPrice
    float loPoint  = useLogScaleInput ? toLog(loPrice) : loPrice
    float fibPrice = (
        isBull and invertFibInput or not isBull and not invertFibInput ?
        loPoint + ((hiPoint - loPoint) * value) :
        hiPoint - ((hiPoint - loPoint) * value)
    )
    float result = useLogScaleInput ? fromLog(fibPrice) : fibPrice


// @function            Selects a line style based on a specified style string.
// @param styleString   (series string) The value of one of the `TY*` variables declared above.
// @returns             (string) The value of a `line.style*` constant.
lineStyle(simple string styleString) =>
    string result = switch styleString
        TY01 => line.style_solid
        TY02 => line.style_dashed
        TY03 => line.style_dotted


// @function            Populates global arrays with a new `fibLevel` ID and a Fibonacci value if the specified
//                      condition allows it.
// @param condition     (simple bool) If `true`, the function creates a new `fibLevel` object, adds its ID to the
//                      global `fibLevels` array, and adds the specified value to the `levels` array. If `false`, the
//                      function ignores the level and does not create a new object.
// @param value         (simple float) The Fibonacci ratio of the level.
// @param color         (series color) The color of the fib lines and label.
// @returns             (void) The function does not return a usable value.
populate(simple bool condition, simple float value, series color color) =>
    if condition
        fibLevel fib = fibLevel.new(value, color)
        fib.fibLine := line.new(
            na, na, na, na, xloc.bar_time, extendFib, color, lineStyle(fibStyleInput), levelsWidthInput
        )
        if extendRtInput
            fib.extLine := line.new(
                na, na, na, na, xloc.bar_time, extendExt, color, lineStyle(extStyleInput), levelsWidthInput
            )
        if showLabels
            fib.fibLabel := label.new(
                na, na, "", xloc.bar_time, yloc.price, color(na), labelStyle, color, labelSizeInput
            )
        array.push(levels, value)
        array.push(fibLevels, fib)


// @function            Creates `linefill` objects to color the space between the lines referenced by `fibLevel` objects.
// @returns             (void) The function does not return a usable value.
setBg() =>
    line temp1 = na
    line temp2 = na
    for sortedIndex in sortedArray
        fibLevel fib = array.get(fibLevels, sortedIndex)
        color fillColor = color.new(fib.fibColor, bgTranspInput)
        if not na(temp1)
            linefill.new(temp1, fib.fibLine, fillColor)
            if extendRtInput
                linefill.new(temp2, fib.extLine, fillColor)
        temp1 := fib.fibLine
        temp2 := fib.extLine


// @function            Constructs a string for display in a fib label. The string contains the level's price and/or
//                      the level's Fibonacci ratio or percentage, depending on the "Prices" and "Levels" inputs.
// @param value         (series float) The level's Fibonacci ratio.
// @param price         (series float) The price value of the fib level.
// @returns             (string) A string containing information about a fib level.
labelText(series float value, series float price) =>
    string levelString = showPercent ? str.tostring(value * 100, format.percent) : str.format("{0, number, #.###}", value)
    string result = switch
        showPricesInput and showLevelsInput => str.format("{0}({1})", levelString, str.tostring(price, format.mintick))
        showPricesInput                     => str.format("({0})", str.tostring(price, format.mintick))
        showLevelsInput                     => levelString


// @function            Calculates the UNIX timestamp for a fib label based on the "Fib labels" and
//                      "Extend to real time" inputs.
// @param               (series int) The leftmost timestamp of the Fibonacci retracement.
// @param               (series int) The rightmost timestamp of the Fibonacci retracement.
// @returns             (int) The UNIX timestamp to use as the label's x-coordinate.
labelTime(series int leftTime, series int rightTime) =>
    int result = switch labelXInput
        LB01 => leftTime
        LB02 => int(math.avg(leftTime, extendRtInput ? chart.right_visible_bar_time : rightTime))
        LB03 => extendRtInput ? chart.right_visible_bar_time : rightTime


// @function            Creates a label with formatted time and price values, and updates the label's location and text
//                      on each bar.
// @param x             (series int) The UNIX timestamp to use as the label's x-coordinate.
// @param y             (series float) The price value to use as the label's y-coordinate.
// @param style         (series string) Sets the label's style. Accepts one of the `label.style_*` constants.
// @returns             (void) The function does not return a usable value.
timeLabel(series int x, series float y, simple color color, series string style) =>
    var label timeLabel = label.new(
        na, na, na, xloc.bar_time, yloc.price, color.new(color, 80), style, color, tLabelSizeInput
    )
    label.set_xy(timeLabel, x, y)
    label.set_text(
        timeLabel, str.format("{0}\n{1}", str.tostring(y, format.mintick), str.format_time(x, "dd/MM/yy @ HH:mm:ss"))
    )


// @function            Creates a line with input-based properties for the trend line display, and updates the line's
//                      coordinates on each bar.
// @param x1            (series int) The UNIX timestamp of the first point on the line.
// @param y1            (series float) The price value of the first point on the line.
// @param x1            (series int) The UNIX timestamp of the second point on the line.
// @param y1            (series float) The price value of the second point on the line.
// @returns             (void) The function does not return a usable value.
hiLoLine(series int x1, series float y1, series int x2, series float y2) =>
    var line hiLoLine = line.new(
        na, na, na, na, xloc.bar_time, extend.none, trendColorInput, lineStyle(trendStyleInput), trendWidthInput
    )
    line.set_xy1(hiLoLine, x1, y1)
    line.set_xy2(hiLoLine, x2, y2)


// @function            Updates the lines and labels for each `fibLevel` object referenced by the `fibLevels` array.
// @param leftTime      (series int) The leftmost UNIX timestamp of the fib retracement.
// @param rightTime     (series int) The rightmost UNIX timestamp of the fib retracement.
// @param hiPrice       (series float) The high price of the fib retracement.
// @param loPrice       (series float) The low price of the fib retracement.
// @param isBull        (series bool) If `true`, the retracement direction is bullish. If `false`, it is bearish.
// @returns             (void) The function does not return a usable value.
setLevels(series int leftTime, series int rightTime, series float hiPrice, series float loPrice, series bool isBull) =>
    int labelTime = labelTime(leftTime, rightTime)
    for fib in fibLevels
        float fibPrice = getPrice(fib.level, hiPrice, loPrice, isBull)
        line.set_xy1(fib.fibLine, leftTime,  fibPrice)
        line.set_xy2(fib.fibLine, rightTime, fibPrice)
        if extendRtInput
            line.set_xy1(fib.extLine, rightTime, fibPrice)
            line.set_xy2(fib.extLine, chart.right_visible_bar_time, fibPrice)
        if showLabels
            label.set_xy(fib.fibLabel, labelTime, fibPrice)
            label.set_text(fib.fibLabel, labelText(fib.level, fibPrice))


// @function            Adds a new element to the beginning of an `id` array, then removes the element at the end if
//                      the array's size exceeds the specified maximum.
// @param id            (any array type) The ID of an array.
// @param max           (simple int) The maximum size of the array.
// @param value         (series <type of the array's elements>) The value or reference to add to the start of the array.
// @returns             (<type of the array's elements>) The removed element if the array's size reached the maximum,
//                      and `na` otherwise.
addToArray(id, simple int max, value) =>
    array.unshift(id, value)
    if array.size(id) > max
        array.pop(id)


// @function            Creates a new `pivotPoint` object and adds its ID to the specified array after each detected
//                      directional change. If the "Show Zig Zag" input is enabled, it also draws a line connecting
//                      the current and previous pivot points.
// @param length        (simple int) The number of bars to search to detect pivot points.
// @param pivotArray    (array<pivotPoint>) The ID of an array containing `pivotPoint` references.
// @param addPivots     (simple bool) If `true`, the function automatically adds a pivot point in the opposite direction
//                      if it does not find a new pivot over `length` bars. Otherwise, it waits for a new pivot.
//                      The default is `true`.
// @returns             (pivotPoint) The `pivotPoint` ID removed from the array if the array's size reached the maximum
//                      size based on the `pivotInput` value, and `na` otherwise.
addPivot(simple int length, array<pivotPoint> pivotArray, simple bool addPivots = true) =>
    pivotPoint piv  = array.get(pivotArray, 0)
    int   altLen    = math.max(piv.bars, 1)
    int   hiBar     = ta.highestbars(length)
    int   loBar     = ta.lowestbars(length)
    int   altHiBar  = ta.highestbars(altLen)
    int   altLoBar  = ta.lowestbars(altLen)
    float altHi     = ta.highest(altLen)
    float altLo     = ta.lowest(altLen)
    bool  missedPiv = piv.bars >= length and addPivots and not
         (hiBar == 0 and high > piv.end.y or loBar == 0 and low < piv.end.y)
    var int dir = na
    dir := switch
        missedPiv => nz(dir, altHiBar < altLoBar ? 1 : -1) * -1
        hiBar + loBar == 0 => dir * -1
        hiBar == 0 =>  1
        loBar == 0 => -1
        => dir
    if dir != dir[1] and time <= chart.right_visible_bar_time
        [hiLo, altHiLo, offset, lineColor] = switch
            dir > 0 => [high, altHi, altHiBar, zzBullColorInput]
            =>         [low,  altLo, altLoBar, zzBearColorInput]
        [pivY, bars] = switch
            missedPiv => [altHiLo, math.abs(offset)]
            =>           [hiLo, 0]
        point newPoint   = point.new(pivY, time[bars])
        pivotPoint pivot = pivotPoint.new(piv.end, newPoint, dir, bars = bars)
        if showZZInput and time >= chart.left_visible_bar_time
            pivot.ln := line.new(piv.end.x, piv.end.y, time[bars], pivY,
              xloc.bar_time, extend.none, lineColor, lineStyle(zzStyleInput), zzwidthInput)
        addToArray(pivotArray, math.max(3, pivotInput + 1), pivot)


// @function            Updates the coordinates of a developing pivot point if the latest calculated point's
//                      coordinates exceed the last saved ones.
// @param length        (simple int) The number of bars to search to detect pivot points.
// @param pivotArray    (array<pivotPoint>) The ID of an array containing `pivotPoint` references.
// @returns             (void) The function does not return a usable value.
updatePivot(simple int length, array<pivotPoint> pivotArray) =>
    if time <= chart.right_visible_bar_time
        pivotPoint piv = array.get(pivotArray, 0)
        int   dir  = piv.dir
        float hiLo = dir > 0 ? high : low
        piv.bars  += 1
        if hiLo * dir > piv.end.y * dir
            piv.end.y := hiLo
            piv.end.x := time
            piv.bars  := 0
            if showZZInput and time >= chart.left_visible_bar_time
                if na(piv.ln)
                    color lineColor = dir > 0 ? zzBullColorInput : zzBearColorInput
                    piv.ln := line.new(piv.start.x, piv.start.y, time, hiLo,
                     xloc.bar_time, extend.none, lineColor, lineStyle(zzStyleInput), zzwidthInput)
            line.set_xy2(piv.ln, time, hiLo)


// @function            Calculates a Zig Zag based on pivots formed over the specified lookback length.
// @param length        (simple int) The number of bars to search to detect pivot points.
// @param addPivots     (simple bool) If `true`, the function automatically adds a pivot point in the opposite direction
//                      if it does not find a new pivot over `length` bars. Otherwise, it waits for a new pivot.
//                      The default is `true`.
// @returns             (array<pivotPoint>) The ID of an array whose elements refer to `pivotPoint` objects that
//                      define the Zig Zag's structure and visuals.
zigZag(simple int length, simple bool addPivots = true) =>
    var array<pivotPoint> pivots = array.new<pivotPoint>(1, pivotPoint.new(point.new(close), point.new(close)))
    updatePivot(length, pivots)
    addPivot(length, pivots, addPivots)
    pivots


// @function            Retrieves the coordinates of the Zig Zag trend line that is N pivots back in the chart's visible
//                      range, and sets the style of the line representing that trend line if the "Show Zig Zag" input
//                      is enabled.
// @param nthPivot      (simple int) The number of pivots back, relative to the latest visible pivot.
// @returns             ([int, float, int, float]) A tuple containing the coordinates of the left and right pivot points
//                      of the Zig Zag line.
findHiLo(simple int nthPivot) =>
    int   leftX  = na
    float leftY  = na
    int   rightX = na
    float rightY = na
    if isZigZag
        array<pivotPoint> pivotArray = zigZag(lengthInput, addPivotInput)
        if barstate.islast
            pivotPoint pivot = array.get(pivotArray, nthPivot)
            leftX  := pivot.start.x
            leftY  := pivot.start.y
            rightX := pivot.end.x
            rightY := pivot.end.y
            if showZZInput
                line.set_style(pivot.ln, lineStyle(trendStyleInput))
    [leftX, leftY, rightX, rightY]
