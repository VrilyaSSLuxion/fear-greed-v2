oscRaw(simple int length, simple bool wait) =>
    float hh = ta.highest(high, length)
    float ll = ta.lowest(low, length)
    float d  = hh - ll
    float v  = d > 0 ? (close - ll) / d * 100 - 100 : -50.0
    wait ? v[1] : v

    request.security(syminfo.tickerid, tf, oscRaw(len, wait), lookahead=barmerge.lookahead_off)

detectSignals(float osc, float upper, float lower) =>
    var int   nextSig = 0
    var float seqTop  = -105.0
    var float seqBot  = 5.0
    var float peakTop = na
    var float peakBot = na
    var bool  inTop   = false
    var bool  inBot   = false
    bool trigTop = false
    bool trigBot = false

    if not na(osc)
        if osc > upper
            if not inTop
                inTop   := true
                peakTop := osc
            else if osc > peakTop
                peakTop := osc
        else if inTop
            if not na(peakTop) and peakTop > upper
                if nextSig == 0 or nextSig == 1 or (nextSig == -1 and peakTop > seqTop)
                    trigTop := true
                    seqTop  := peakTop
                    nextSig := -1
            inTop   := false
            peakTop := na

        if osc < lower
            if not inBot
                inBot   := true
                peakBot := osc
            else if osc < peakBot
                peakBot := osc
        else if inBot
            if not na(peakBot) and peakBot < lower
                if nextSig == 0 or nextSig == -1 or (nextSig == 1 and peakBot < seqBot)
                    trigBot := true
                    seqBot  := peakBot
                    nextSig := 1
            inBot   := false
            peakBot := na
    [trigTop, trigBot]
