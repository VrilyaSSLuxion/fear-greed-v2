//@version=5
indicator("Buy Low Sell High Indicator", overlay=true)

// Parameter für die Anzahl der Kerzen, auf denen die extremen Werte basieren
length = input.int(14, title="Lookback Period", minval=1)

// Berechnung der niedrigsten und höchsten Kurse der letzten `length` Kerzen
lowestLow = ta.lowest(low, length)
highestHigh = ta.highest(high, length)

// Kauf- und Verkaufssignale
buySignal = low == lowestLow
sellSignal = high == highestHigh

// Plotte die Kauf- und Verkaufssignale auf dem Chart
plotshape(series=buySignal, title="Buy Signal", location=location.belowbar, color=color.green, style=shape.labelup, text="Buy", textcolor=color.white, size=size.small)
plotshape(series=sellSignal, title="Sell Signal", location=location.abovebar, color=color.red, style=shape.labeldown, text="Sell", textcolor=color.white, size=size.small)

// Alarme
alertcondition(buySignal, title="Buy Signal Alert", message="Buy signal generated: Price is at the lowest of the last period.")
alertcondition(sellSignal, title="Sell Signal Alert", message="Sell signal generated: Price is at the highest of the last period.")
