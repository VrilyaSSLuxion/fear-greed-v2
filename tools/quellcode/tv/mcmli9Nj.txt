//@version=5
indicator(title="Relative Strength Index", shorttitle="RSI", format=format.price, precision=2, timeframe="", timeframe_gaps=true)

// Colors
color rsi_color = color.yellow
color ma_color = color.red
color upper_band_color = color.red
color lower_band_color = color.green
color upper_band_2_color = color.red
color middle_band_2_color = color.white
color lower_band_2_color = color.green

color upper_band_3_color = color.red
color middle_band_3_color = color.white
color lower_band_3_color = color.green

color background_fill_color = color.rgb(126, 87, 194, 90)
color overbought_color = color.new(color.green, 0)
color oversold_color = color.new(color.red, 100)
color none_color = color.new(color.white, 100)
color bear_color = color.red
color bull_color = color.green
color text_color = color.white

ma(source, length, type) =>
    switch type
        "SMA" => ta.sma(source, length)
        "Bollinger Bands" => ta.sma(source, length)
        "EMA" => ta.ema(source, length)
        "SMMA (RMA)" => ta.rma(source, length)
        "WMA" => ta.wma(source, length)
        "VWMA" => ta.vwma(source, length)

rsiLengthInput = input.int(14, minval=1, title="RSI Length-dafault-14-", group="RSI Settings")
rsiSourceInput = input.source(ohlc4, "Source", group="RSI Settings")
maTypeInput = input.string("WMA", title="MA Type-default-SMA-", options=["SMA", "Bollinger Bands", "EMA", "SMMA (RMA)", "WMA", "VWMA"], group="MA Settings", display = display.data_window)
maLengthInput = input.int(14, title="MA Length", group="MA Settings", display = display.data_window)
bbMultInput = input.float(2.0, minval=0.001, maxval=50, title="BB StdDev", group="MA Settings", display = display.data_window)
showDivergence = input.bool(true, title="Show Divergence", group="RSI Settings", display = display.data_window)

// Custom RSI Bands
upper_band_level = input.int(99, minval=0, maxval=100, title="Upper Band Level", group="RSI Bands")
lower_band_level = input.int(79, minval=0, maxval=100, title="Lower Band Level", group="RSI Bands")

upper_band_2_level = input.int(62, minval=0, maxval=100, title="Upper Band 2 Level", group="RSI Bands")
middle_band_2_level = input.int(50, minval=0, maxval=100, title="Middle Band 2 Level", group="RSI Bands")
lower_band_2_level = input.int(38, minval=0, maxval=100, title="Lower Band 2 Level", group="RSI Bands")

upper_band_3_level = input.int(21, minval=0, maxval=100, title="Upper Band 3 Level", group="RSI Bands")
middle_band_3_level = input.int(11, minval=0, maxval=100, title="Middle Band 3 Level", group="RSI Bands")
lower_band_3_level = input.int(1, minval=0, maxval=100, title="Lower Band 3 Level", group="RSI Bands")


up = ta.rma(math.max(ta.change(rsiSourceInput), 0), rsiLengthInput)
down = ta.rma(-math.min(ta.change(rsiSourceInput), 0), rsiLengthInput)
rsi = down == 0 ? 100 : up == 0 ? 0 : 100 - (100 / (1 + up / down))
rsiMA = ma(rsi, maLengthInput, maTypeInput)
isBB = maTypeInput == "Bollinger Bands"
 
rsiPlot = plot(rsi, "RSI", color=rsi_color)
plot(rsiMA, "RSI-based MA", color=ma_color)
rsiUpperBand = hline(upper_band_level, "RSI Upper Band", color=upper_band_color)
midline = hline(89, "RSI Middle Band", color=color.new(color.white, 0))
rsiLowerBand = hline(lower_band_level, "RSI Lower Band", color=lower_band_color)
rsiUpperBand2 = hline(upper_band_2_level, "RSI Upper Band 2", color=upper_band_2_color)
rsiMiddleBand2 = hline(middle_band_2_level, "RSI Middle Band 2", color=middle_band_2_color)
rsiLowerBand2 = hline(lower_band_2_level, "RSI Lower Band 2", color=lower_band_2_color)
rsiUpperBand3 = hline(upper_band_3_level, "RSI Upper Band 3", color=upper_band_3_color)
rsiMiddleBand3 = hline(middle_band_3_level, "RSI Middle Band 3", color=middle_band_3_color)
rsiLowerBand3 = hline(lower_band_3_level, "RSI Lower Band 3", color=lower_band_3_color)


fill(rsiUpperBand, rsiLowerBand, color=color.new(background_fill_color, 90), title="RSI Background Fill")
fill(rsiUpperBand2, rsiLowerBand2, color=color.new(background_fill_color, 90), title="RSI Background Fill 2")
fill(rsiUpperBand3, rsiLowerBand3, color=color.new(background_fill_color, 90), title="RSI Background Fill 3")


midLinePlot = plot(50, color = na, editable = false, display = display.none)
fill(rsiPlot, midLinePlot, 100, upper_band_level, top_color=overbought_color, bottom_color=color.new(overbought_color, 100), title="Overbought Gradient Fill")
fill(rsiPlot, midLinePlot, lower_band_level, 0, top_color=color.new(oversold_color, 100), bottom_color=oversold_color, title="Oversold Gradient Fill")
fill(rsiPlot, midLinePlot, 100, upper_band_2_level, top_color=color.new(overbought_color, 50), bottom_color=color.new(overbought_color, 79), title="Overbought Gradient Fill 2")
fill(rsiPlot, midLinePlot, lower_band_2_level, 0, top_color=color.new(oversold_color, 21), bottom_color=color.new(oversold_color, 50), title="Oversold Gradient Fill 2")

fill(rsiPlot, midLinePlot, 100, upper_band_3_level, top_color=color.new(overbought_color, 50), bottom_color=color.new(overbought_color, 79), title="Overbought Gradient Fill 3")
fill(rsiPlot, midLinePlot, lower_band_3_level, 0, top_color=color.new(oversold_color, 21), bottom_color=color.new(oversold_color, 50), title="Oversold Gradient Fill 3")



// Divergence
lookbackRight = 5
lookbackLeft = 5
rangeUpper = 60
rangeLower = 5

plFound = na(ta.pivotlow(rsi, lookbackLeft, lookbackRight)) ? false : true
phFound = na(ta.pivothigh(rsi, lookbackLeft, lookbackRight)) ? false : true

_inRange(cond) =>
    bars = ta.barssince(cond == true)
    rangeLower <= bars and bars <= rangeUpper

// Regular Bullish
rsiHL = rsi[lookbackRight] > ta.valuewhen(plFound, rsi[lookbackRight], 1) and _inRange(plFound[1])
priceLL = low[lookbackRight] < ta.valuewhen(plFound, low[lookbackRight], 1)
bullCondAlert = priceLL and rsiHL and plFound
bullCond = showDivergence and bullCondAlert

plot(plFound ? rsi[lookbackRight] : na, offset=-lookbackRight, title="Regular Bullish", linewidth=2, color=(bullCond ? bull_color : none_color), display = display.pane)
plotshape(bullCond ? rsi[lookbackRight] : na, offset=-lookbackRight, title="Regular Bullish Label", text=" Bull ", style=shape.labelup, location=location.absolute, color=bull_color, textcolor=text_color)

// Regular Bearish
rsiLH = rsi[lookbackRight] < ta.valuewhen(phFound, rsi[lookbackRight], 1) and _inRange(phFound[1])
priceHH = high[lookbackRight] > ta.valuewhen(phFound, high[lookbackRight], 1)
bearCondAlert = priceHH and rsiLH and phFound
bearCond = showDivergence and bearCondAlert

plot(phFound ? rsi[lookbackRight] : na, offset=-lookbackRight, title="Regular Bearish", linewidth=2, color=(bearCond ? bear_color : none_color), display = display.pane)
plotshape(bearCond ? rsi[lookbackRight] : na, offset=-lookbackRight, title="Regular Bearish Label", text=" Bear ", style=shape.labeldown, location=location.absolute, color=bear_color, textcolor=text_color)

alertcondition(bullCondAlert, title='Regular Bullish Divergence', message="Found a new Regular Bullish Divergence, `Pivot Lookback Right` number of bars to the left of the current bar.")
alertcondition(bearCondAlert, title='Regular Bearish Divergence', message='Found a new Regular Bearish Divergence, `Pivot Lookback Right` number of bars to the left of the current bar.')
