//@version=5
indicator("-VrilyaSS-TV-BB+MACD-BS-(HA-Candles-2 Min-Chart/Trades Time)-", overlay=true)

// Inputs für Bollinger Bands
bbLength = input.int(6, title="Bollinger Bands Length-6-")
bbMultiplier = input.float(1.3, title="Bollinger Bands Multiplier-1,3-")
src = input(close, title="Source")

maType = input.string("SMA", "Basis MA Type", options=["SMA", "EMA", "SMMA (RMA)", "WMA", "VWMA"])

ma(source, length, _type) =>
    switch _type
        "SMA" => ta.sma(source, length)
        "EMA" => ta.ema(source, length)
        "SMMA (RMA)" => ta.rma(source, length)
        "WMA" => ta.wma(source, length)
        "VWMA" => ta.vwma(source, length)

basis = ma(src, bbLength, maType)
dev = bbMultiplier * ta.stdev(src, bbLength)
upper = basis + dev
lower = basis - dev
offset = input.int(0, "Offset", minval=-500, maxval=500)

p1 = plot(upper, "Upper", color=color.white, linewidth=2, offset=offset)
p2 = plot(lower, "Lower", color=color.white, linewidth=2, offset=offset)
p3 = plot(basis, "Basis", color=#000000, linewidth=2, offset=offset)
fill(p1, p2, title="Background", color=color.new(color.green, 85))

// Inputs für MACD
macdFastLength = input.int(6, title="MACD Fast Length-6-")
macdSlowLength = input.int(19, title="MACD Slow Length-19-")
macdSignalLength = input.int(6, title="MACD Signal Length-6-")
[macdLine, macdSignal, _] = ta.macd(close, macdFastLength, macdSlowLength, macdSignalLength)

// MACD-Kreuzungen
macdCrossUp = ta.crossover(macdLine, macdSignal)
macdCrossDown = ta.crossunder(macdLine, macdSignal)

// MACD über und unter Null-Linie
macdBelowZero = macdLine < 0
macdAboveZero = macdLine > 0

// Bedingungen für Bollinger Bands
nearUpperBand = close > upper * 0.95
nearLowerBand = close < lower * 1.05

// Signale
buySignal = macdCrossUp and macdBelowZero and (close >= upper or nearUpperBand)
sellSignal = macdCrossDown and macdAboveZero and (close <= lower or nearLowerBand)

// Plotte Kauf- und Verkaufssignale
plotshape(series=buySignal, title="Buy Signal", location=location.belowbar, color=color.green, style=shape.labelup, text="Buy", textcolor=color.white, size=size.small)
plotshape(series=sellSignal, title="Sell Signal", location=location.abovebar, color=color.red, style=shape.labeldown, text="Sell", textcolor=color.white, size=size.small)

// Alarme
alertcondition(buySignal, title="Buy Signal Alert", message="Buy signal generated based on Bollinger Bands and MACD.")
alertcondition(sellSignal, title="Sell Signal Alert", message="Sell signal generated based on Bollinger Bands and MACD.")
alertcondition(buySignal or sellSignal, title="Buy/Sell Signal Alert", message="Buy or Sell signal generated based on Bollinger Bands and MACD.")

// Plotte MACD standardmäßig deaktiviert
macdVisible = input(false, title="Show MACD Lines")
plot(macdLine, title="MACD Line", color=color.green, linewidth=2, display=macdVisible ? display.all : display.none)
plot(macdSignal, title="MACD Signal Line", color=color.red, linewidth=2, display=macdVisible ? display.all : display.none)
