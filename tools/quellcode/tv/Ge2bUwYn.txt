// Funktionen zur Überprüfung der Muster
isHammer(o, h, l, c) =>
    bodySize = math.abs(c - o)
    lowerShadow = math.min(o, c) - l
    upperShadow = h - math.max(o, c)
    isGreen = c > o
    lowerShadow > 2 * bodySize and upperShadow < bodySize and isGreen

isInvertedHammer(o, h, l, c) =>
    bodySize = math.abs(c - o)
    lowerShadow = math.min(o, c) - l
    upperShadow = h - math.max(o, c)
    isGreen = c > o
    upperShadow > 2 * bodySize and lowerShadow < bodySize and isGreen

isDragonflyDoji(o, h, l, c) =>
    bodySize = math.abs(c - o)
    lowerShadow = math.min(o, c) - l
    upperShadow = h - math.max(o, c)
    isGreen = c > o
    bodySize < (h - l) * 0.1 and lowerShadow > (h - l) * 0.4 and upperShadow < (h - l) * 0.1 and isGreen

isBullishSpinningTop(o, h, l, c) =>
    bodySize = math.abs(c - o)
    totalRange = h - l
    upperShadow = h - math.max(o, c)
    lowerShadow = math.min(o, c) - l
    isGreen = c > o
    bodySize < totalRange * 0.3 and upperShadow > bodySize and lowerShadow > bodySize and isGreen

isHangingMan(o, h, l, c) =>
    bodySize = math.abs(c - o)
    lowerShadow = math.min(o, c) - l
    upperShadow = h - math.max(o, c)
    isRed = c < o
    lowerShadow > 2 * bodySize and upperShadow < bodySize and isRed

isShootingStar(o, h, l, c) =>
    bodySize = math.abs(c - o)
    lowerShadow = math.min(o, c) - l
    upperShadow = h - math.max(o, c)
    isRed = c < o
    upperShadow > 2 * bodySize and lowerShadow < bodySize and isRed

isGravestoneDoji(o, h, l, c) =>
    bodySize = math.abs(c - o)
    lowerShadow = math.min(o, c) - l
    upperShadow = h - math.max(o, c)
    isRed = c < o
    bodySize < (h - l) * 0.1 and lowerShadow < (h - l) * 0.1 and upperShadow > (h - l) * 0.4 and isRed

isBearishSpinningTop(o, h, l, c) =>
    bodySize = math.abs(c - o)
    totalRange = h - l
    upperShadow = h - math.max(o, c)
    lowerShadow = math.min(o, c) - l
    isRed = c < o
    bodySize < totalRange * 0.3 and upperShadow > bodySize and lowerShadow > bodySize and isRed
