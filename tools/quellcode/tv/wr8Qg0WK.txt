//@version=5
indicator("1-Minute RSI Bollinger Band Indicator", overlay=true)

// Input settings
rsiLength = input.int(14, title="RSI Length")
rsiSource = input.source(close, title="RSI Source", defval=hlc3)
rsiOverbought = input.int(70, title="RSI Overbought Level", minval=50, maxval=100)
rsiOversold = input.int(30, title="RSI Oversold Level", minval=0, maxval=50)
bollingerLength = input.int(20, title="Bollinger Bands Length")
bollingerMultiplier = input.float(2.0, title="Bollinger Bands Multiplier")

// RSI Calculation
rsi = ta.rsi(rsiSource, rsiLength)

// Bollinger Bands Calculation
basis = ta.sma(close, bollingerLength)
dev = bollingerMultiplier * ta.stdev(close, bollingerLength)
upperBand = basis + dev
lowerBand = basis - dev

// Buy Condition
buyCondition = ta.crossover(close, lowerBand) and rsi < rsiOversold

// Sell Condition
sellCondition = ta.crossunder(close, upperBand) and rsi > rsiOverbought

// Plot Buy and Sell Signals
plotshape(series=buyCondition, location=location.belowbar, color=color.new(color.green, 0), style=shape.labelup, text="BUY", textcolor=color.black, size=size.tiny)
plotshape(series=sellCondition, location=location.abovebar, color=color.new(color.red, 0), style=shape.labeldown, text="SELL", textcolor=color.black, size=size.tiny)

// Alerts
alertcondition(buyCondition, title="Buy Signal", message="Buy Signal: Price crosses above the lower Bollinger Band and RSI is below the oversold level.")
alertcondition(sellCondition, title="Sell Signal", message="Sell Signal: Price crosses below the upper Bollinger Band and RSI is above the overbought level.")

// Plot RSI and Bollinger Bands for reference
plot(rsi, title="RSI", color=color.yellow, linewidth=2, display=display.none)
hline(rsiOversold, "RSI Oversold Level", color=color.green, linewidth=2, linestyle=hline.style_solid, display=display.none)
hline(rsiOverbought, "RSI Overbought Level", color=color.red, linewidth=2, linestyle=hline.style_solid, display=display.none)
plot(upperBand, title="Upper Bollinger Band", color=color.red, linewidth=2)
plot(basis, title="Middle Bollinger Band", color=color.white, linewidth=2)
plot(lowerBand, title="Lower Bollinger Band", color=color.green, linewidth=2)
