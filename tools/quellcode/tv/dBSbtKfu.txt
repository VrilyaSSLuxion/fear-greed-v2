//@version=5
indicator("Alternative Short-Term Trade Indicator", overlay=true)

// Input settings
macdFastLength = input.int(12, title="MACD Fast Length")
macdSlowLength = input.int(26, title="MACD Slow Length")
macdSignalSmoothing = input.int(9, title="MACD Signal Smoothing")
rsiLength = input.int(14, title="RSI Length")
rsiSource = input.source(ohlc4, title="RSI Source")
rsiOverbought = input.int(70, title="RSI Overbought Level", minval=50, maxval=100)
rsiOversold = input.int(30, title="RSI Oversold Level", minval=0, maxval=50)

// Calculations
[macdLine, signalLine, _] = ta.macd(close, macdFastLength, macdSlowLength, macdSignalSmoothing)
rsi = ta.rsi(rsiSource, rsiLength)

// Buy Condition
buyCondition = ta.crossover(macdLine, signalLine) and rsi < rsiOversold

// Sell Condition
sellCondition = ta.crossunder(macdLine, signalLine) and rsi > rsiOverbought

// Plot Buy and Sell Signals with small labels
var label buyLabel = na
var label sellLabel = na
buyLabel := label.new(na, na, text="BUY", color=color.new(color.green, 0), style=label.style_label_up, size=size.tiny)
sellLabel := label.new(na, na, text="SELL", color=color.new(color.red, 0), style=label.style_label_down, size=size.tiny)

// Plot Buy and Sell Conditions
plotshape(series=buyCondition, location=location.belowbar, color=color.new(color.green, 0), style=shape.labelup, text="", size=size.tiny)
plotshape(series=sellCondition, location=location.abovebar, color=color.new(color.red, 0), style=shape.labeldown, text="", size=size.tiny)

// Alerts
alertcondition(buyCondition, title="Buy Signal", message="Buy Signal: MACD crosses above Signal Line and RSI below oversold level.")
alertcondition(sellCondition, title="Sell Signal", message="Sell Signal: MACD crosses below Signal Line and RSI above overbought level.")

// Plot MACD and Signal Line for reference (default visibility off)
plot(macdLine, title="MACD Line", color=color.blue, linewidth=2, display=display.none)
plot(signalLine, title="Signal Line", color=color.orange, linewidth=2, display=display.none)
hline(rsiOversold, "RSI Oversold Level", color=color.green, linewidth=2, linestyle=hline.style_solid, display=display.none)
hline(rsiOverbought, "RSI Overbought Level", color=color.red, linewidth=2, linestyle=hline.style_solid, display=display.none)
