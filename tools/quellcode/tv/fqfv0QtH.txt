//@version=5
indicator("-VrilyaSS-TC-Triple Candle Patterns-small-", overlay=true)

// Funktionen zur Überprüfung der Muster
isBullishMorningStar(o1, c1, o2, c2, o3, c3) =>
    c1 < o1 and c2 < o2 and c3 > o3 and c3 > (o1 + c1) / 2 and o2 < c1 and o2 > c2

isBullishAbandonedBaby(o1, c1, h2, l2, o3, c3) =>
    c1 < o1 and c3 > o3 and l2 > c1 and h2 < o3

isBullishThreeWhiteSoldiers(o1, c1, o2, c2, o3, c3) =>
    c1 > o1 and c2 > o2 and c3 > o3 and o2 > c1 and o3 > c2

isBullishThreeLineStrike(o1, c1, o2, c2, o3, c3, o4, c4) =>
    c1 > o1 and c2 > o2 and c3 > o3 and c4 < o4 and c4 < o3 and c4 > o1

isBullishMorningDojiStar(o1, c1, o2, c2, o3, c3) =>
    c1 < o1 and o2 == c2 and c3 > o3 and c3 > (o1 + c1) / 2 and o2 < c1

isBearishAbandonedBaby(o1, c1, h2, l2, o3, c3) =>
    c1 > o1 and c3 < o3 and h2 < c1 and l2 > o3

isBearishThreeBlackCrows(o1, c1, o2, c2, o3, c3) =>
    c1 < o1 and c2 < o2 and c3 < o3 and o2 < c1 and o3 < c2

isBearishEveningDojiStar(o1, c1, o2, c2, o3, c3) =>
    c1 > o1 and o2 == c2 and c3 < o3 and c3 < (o1 + c1) / 2 and o2 > c1

isBearishEveningStar(o1, c1, o2, c2, o3, c3) =>
    c1 > o1 and c2 > o2 and c3 < o3 and c3 < (o1 + c1) / 2 and o2 > c1

// Checkboxen und Beschreibungen
showBullishMorningStar = input(true, title="Bullish Morning Star anzeigen")
showBullishAbandonedBaby = input(true, title="Bullish Abandoned Baby anzeigen")
showBullishThreeWhiteSoldiers = input(true, title="Bullish Three White Soldiers anzeigen")

showBearishAbandonedBaby = input(true, title="Bearish Abandoned Baby anzeigen")

// Beschreibungs-Labels

var label bullishMorningDojiStarLabel = na

    bullishMorningStarLabel := label.new(bar_index, high, text="Zeigt einen Bullish Morning Star an.", xloc=xloc.bar_index, yloc=yloc.abovebar, style=label.style_none, textcolor=color.black, size=size.small)
    bullishAbandonedBabyLabel := label.new(bar_index, high, text="Zeigt ein Bullish Abandoned Baby an.", xloc=xloc.bar_index, yloc=yloc.abovebar, style=label.style_none, textcolor=color.black, size=size.small)
    bullishThreeWhiteSoldiersLabel := label.new(bar_index, high, text="Zeigt drei weiße Soldaten an.", xloc=xloc.bar_index, yloc=yloc.abovebar, style=label.style_none, textcolor=color.black, size=size.small)
    bullishThreeLineStrikeLabel := label.new(bar_index, high, text="Zeigt einen Bullish Three Line Strike an.", xloc=xloc.bar_index, yloc=yloc.abovebar, style=label.style_none, textcolor=color.black, size=size.small)
    bullishMorningDojiStarLabel := label.new(bar_index, high, text="Zeigt einen Bullish Morning Doji Star an.", xloc=xloc.bar_index, yloc=yloc.abovebar, style=label.style_none, textcolor=color.black, size=size.small)

    bearishAbandonedBabyLabel := label.new(bar_index, high, text="Zeigt ein Bearish Abandoned Baby an.", xloc=xloc.bar_index, yloc=yloc.abovebar, style=label.style_none, textcolor=color.black, size=size.small)
    bearishThreeBlackCrowsLabel := label.new(bar_index, high, text="Zeigt drei schwarze Krähen an.", xloc=xloc.bar_index, yloc=yloc.abovebar, style=label.style_none, textcolor=color.black, size=size.small)
    bearishEveningDojiStarLabel := label.new(bar_index, high, text="Zeigt einen Bearish Evening Doji Star an.", xloc=xloc.bar_index, yloc=yloc.abovebar, style=label.style_none, textcolor=color.black, size=size.small)
    bearishEveningStarLabel := label.new(bar_index, high, text="Zeigt einen Bearish Evening Star an.", xloc=xloc.bar_index, yloc=yloc.abovebar, style=label.style_none, textcolor=color.black, size=size.small)

// Berechnung der Muster
bullishMorningStar = isBullishMorningStar(open[2], close[2], open[1], close[1], open, close)
bullishAbandonedBaby = isBullishAbandonedBaby(open[2], close[2], high[1], low[1], open, close)
bullishThreeWhiteSoldiers = isBullishThreeWhiteSoldiers(open[2], close[2], open[1], close[1], open, close)
bullishThreeLineStrike = isBullishThreeLineStrike(open[3], close[3], open[2], close[2], open[1], close[1], open, close)
bullishMorningDojiStar = isBullishMorningDojiStar(open[2], close[2], open[1], close[1], open, close)

bearishAbandonedBaby = isBearishAbandonedBaby(open[2], close[2], high[1], low[1], open, close)
bearishThreeBlackCrows = isBearishThreeBlackCrows(open[2], close[2], open[1], close[1], open, close)
bearishEveningDojiStar = isBearishEveningDojiStar(open[2], close[2], open[1], close[1], open, close)
bearishEveningStar = isBearishEveningStar(open[2], close[2], open[1], close[1], open, close)

// Plots für Alarme

plotshape(series=showBearishEveningStar and bearishEveningStar, location=location.abovebar, color=color.red, style=shape.labeldown, text="ES", title="Bearish Evening Star", textcolor=color.black, size=size.small)
