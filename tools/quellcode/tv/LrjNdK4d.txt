//@version=5
// =====================================================================
//  PEAK DECODER - Multi-Timeframe Oszillator
//  Wertebereich 0 bis -100 (Williams %R). Alle Werte negativ.
//  LONG-Extrem  bei -89   ->  gruener Pfeil UNTER der Fraktalkerze
//  SHORT-Extrem bei -11   ->  roter  Pfeil UEBER der Fraktalkerze
// =====================================================================
indicator("Peak Decoder MTF", shorttitle = "PeakDecoder", overlay = false,
          max_lines_count = 100, max_labels_count = 200)

// ---------------------------------------------------------------------
// HINWEIS: Die Periodenlaenge darf KEINE Funktion sein.
// Funktionen liefern in Pine immer 'series', request.security braucht
// aber 'simple int'. Darum direkt als Ausdruck berechnen.
// ---------------------------------------------------------------------

// ---------------------------------------------------------------------
// INSTANZ 1
// ---------------------------------------------------------------------
g1 = "Instanz 1"
i1On    = input.bool(true,        "Aktiv",       group = g1)
i1TF    = input.timeframe("240",  "Timeframe",   group = g1)
i1Mode  = input.string("Passiv",  "Modus",       options = ["Aggressiv", "Normal", "Passiv"], group = g1)
i1Col   = input.color(color.new(#2962FF, 0), "Farbe", group = g1)
i1Long  = input.float(-89.0, "Long-Level",  minval = -100, maxval = 0, step = 1, group = g1)
i1Short = input.float(-11.0, "Short-Level", minval = -100, maxval = 0, step = 1, group = g1)

// ---------------------------------------------------------------------
// INSTANZ 2
// ---------------------------------------------------------------------
g2 = "Instanz 2"
i2On    = input.bool(false,       "Aktiv",       group = g2)
i2TF    = input.timeframe("30",   "Timeframe",   group = g2)
i2Mode  = input.string("Normal",  "Modus",       options = ["Aggressiv", "Normal", "Passiv"], group = g2)
i2Col   = input.color(color.new(#FF9800, 0), "Farbe", group = g2)
i2Long  = input.float(-85.0, "Long-Level",  minval = -100, maxval = 0, step = 1, group = g2)
i2Short = input.float(-15.0, "Short-Level", minval = -100, maxval = 0, step = 1, group = g2)

// ---------------------------------------------------------------------
// INSTANZ 3
// ---------------------------------------------------------------------
g3 = "Instanz 3"
i3On    = input.bool(false,         "Aktiv",     group = g3)
i3TF    = input.timeframe("15",     "Timeframe", group = g3)
i3Mode  = input.string("Aggressiv", "Modus",     options = ["Aggressiv", "Normal", "Passiv"], group = g3)
i3Col   = input.color(color.new(#F23645, 0), "Farbe", group = g3)
i3Long  = input.float(-80.0, "Long-Level",  minval = -100, maxval = 0, step = 1, group = g3)
i3Short = input.float(-20.0, "Short-Level", minval = -100, maxval = 0, step = 1, group = g3)

// ---------------------------------------------------------------------
// SIGNALE
// ---------------------------------------------------------------------
gs = "Signale"
expiryBars   = input.int(12, "Gueltigkeit des Extrems (Kerzen)", minval = 1, maxval = 100, group = gs)
confirmed    = input.bool(true, "Fraktal erst nach Bestaetigung (kein Repaint)", group = gs)
showExtremes = input.bool(true, "Extrem-Marker im Oszillator", group = gs)

// ---------------------------------------------------------------------
// PERIODENLAENGEN (simple int, keine Funktion!)
// ---------------------------------------------------------------------
i1Len = i1Mode == "Aggressiv" ? 4 : i1Mode == "Passiv" ? 22 : 10
i2Len = i2Mode == "Aggressiv" ? 4 : i2Mode == "Passiv" ? 22 : 10
i3Len = i3Mode == "Aggressiv" ? 4 : i3Mode == "Passiv" ? 22 : 10

// ---------------------------------------------------------------------
// OSZILLATOR: Williams %R auf beliebigem Timeframe
// ---------------------------------------------------------------------
oscOn(simple string tf, simple int len) =>
    request.security(syminfo.tickerid, tf, ta.wpr(len),
                     lookahead = barmerge.lookahead_off)

osc1 = i1On ? oscOn(i1TF, i1Len) : na
osc2 = i2On ? oscOn(i2TF, i2Len) : na
osc3 = i3On ? oscOn(i3TF, i3Len) : na

plot(osc1, "Instanz 1", color = i1Col, linewidth = 2)
plot(osc2, "Instanz 2", color = i2Col, linewidth = 2)
plot(osc3, "Instanz 3", color = i3Col, linewidth = 2)

// ---------------------------------------------------------------------
// ZONEN
// ---------------------------------------------------------------------
upperLine = hline(-11, "Short-Zone", color = color.new(color.red, 30),
                  linestyle = hline.style_dashed)
lowerLine = hline(-89, "Long-Zone",  color = color.new(color.green, 30),
                  linestyle = hline.style_dashed)
hline(-50, "Mitte", color = color.new(color.gray, 70), linestyle = hline.style_dotted)

topBand = hline(0,    display = display.none)
botBand = hline(-100, display = display.none)
fill(upperLine, topBand, color = color.new(color.red,   92))
fill(lowerLine, botBand, color = color.new(color.green, 92))

// ---------------------------------------------------------------------
// EXTREM-ERKENNUNG
// ---------------------------------------------------------------------
longHit  = (i1On and not na(osc1) and osc1 <= i1Long)  or (i2On and not na(osc2) and osc2 <= i2Long)  or (i3On and not na(osc3) and osc3 <= i3Long)
shortHit = (i1On and not na(osc1) and osc1 >= i1Short) or (i2On and not na(osc2) and osc2 >= i2Short) or (i3On and not na(osc3) and osc3 >= i3Short)

var bool longArmed  = false
var bool shortArmed = false
var int  longAge    = 0
var int  shortAge   = 0

if longArmed
    longAge += 1
    if longAge > expiryBars
        longArmed := false

if shortArmed
    shortAge += 1
    if shortAge > expiryBars
        shortArmed := false

if longHit and not longArmed
    longArmed := true
    longAge   := 0

if shortHit and not shortArmed
    shortArmed := true
    shortAge   := 0

plotshape(showExtremes and longHit  and not longArmed[1] ? -95 : na, "Extrem Long",
          style = shape.circle, location = location.absolute,
          color = color.new(color.green, 0), size = size.tiny)
plotshape(showExtremes and shortHit and not shortArmed[1] ? -5 : na, "Extrem Short",
          style = shape.circle, location = location.absolute,
          color = color.new(color.red, 0), size = size.tiny)

// ---------------------------------------------------------------------
// FRAKTAL
//   offs = 1 -> Mittelkerze ist die vorige Kerze (kann repainten)
//   offs = 2 -> Mittelkerze ist bestaetigt
// ---------------------------------------------------------------------
offs = confirmed ? 2 : 1

lowFract  = low[offs]  < low[offs + 1]  and low[offs]  < low[offs - 1]
highFract = high[offs] > high[offs + 1] and high[offs] > high[offs - 1]

longSignal  = longArmed  and lowFract
shortSignal = shortArmed and highFract

if longSignal
    longArmed := false
if shortSignal
    shortArmed := false

// ---------------------------------------------------------------------
// PFEILE AUF DEM KURSCHART
//   Long  -> gruener Pfeil UNTER der Fraktalkerze
//   Short -> roter  Pfeil UEBER der Fraktalkerze
// ---------------------------------------------------------------------
atr = ta.atr(14)

if longSignal
    label.new(bar_index[offs], low[offs] - atr * 0.6,
              text  = "▲",
              color = color.new(color.green, 100),
              textcolor = color.lime,
              style = label.style_label_up,
              size  = size.small,
              yloc  = yloc.price,
              force_overlay = true)

if shortSignal
    label.new(bar_index[offs], high[offs] + atr * 0.6,
              text  = "▼",
              color = color.new(color.red, 100),
              textcolor = color.red,
              style = label.style_label_down,
              size  = size.small,
              yloc  = yloc.price,
              force_overlay = true)

// ---------------------------------------------------------------------
// ALARME
// ---------------------------------------------------------------------
alertcondition(longHit,     "Extrem LONG",  "Peak Decoder: Oszillator im Long-Extrem")
alertcondition(shortHit,    "Extrem SHORT", "Peak Decoder: Oszillator im Short-Extrem")
alertcondition(longSignal,  "Einstieg LONG",  "Peak Decoder: Long-Fraktal bestaetigt")
alertcondition(shortSignal, "Einstieg SHORT", "Peak Decoder: Short-Fraktal bestaetigt")

// ---------------------------------------------------------------------
// INFO-TABELLE
// ---------------------------------------------------------------------
showTable = input.bool(true, "Tabelle anzeigen", group = "Anzeige")

var table t = table.new(position.top_right, 2, 5, border_width = 1, frame_width = 1, frame_color = color.gray)

if showTable and barstate.islast
    txt1 = i1On and not na(osc1) ? str.tostring(osc1, "#.##") : "-"
    txt2 = i2On and not na(osc2) ? str.tostring(osc2, "#.##") : "-"
    txt3 = i3On and not na(osc3) ? str.tostring(osc3, "#.##") : "-"
    setupTxt = longArmed ? "LONG aktiv" : shortArmed ? "SHORT aktiv" : "-"
    setupCol = longArmed ? color.lime : shortArmed ? color.red : color.gray

    table.cell(t, 0, 0, "PEAK DECODER", text_color = color.orange, text_size = size.small)
    table.cell(t, 1, 0, "", text_size = size.small)

    table.cell(t, 0, 1, "I1 " + i1TF, text_color = i1Col, text_size = size.tiny)
    table.cell(t, 1, 1, txt1, text_color = i1Col, text_size = size.tiny)

    table.cell(t, 0, 2, "I2 " + i2TF, text_color = i2Col, text_size = size.tiny)
    table.cell(t, 1, 2, txt2, text_color = i2Col, text_size = size.tiny)

    table.cell(t, 0, 3, "I3 " + i3TF, text_color = i3Col, text_size = size.tiny)
    table.cell(t, 1, 3, txt3, text_color = i3Col, text_size = size.tiny)

    table.cell(t, 0, 4, "Setup", text_color = color.white, text_size = size.tiny)
    table.cell(t, 1, 4, setupTxt, text_color = setupCol, text_size = size.tiny)
