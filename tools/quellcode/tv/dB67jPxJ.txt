//@version=5
indicator(title="QTI with Accelerator Oscillator", shorttitle="QTI+AO", overlay=false)

// Inputs
length = input.int(14, title="Acceleration Factor", minval=1)
qtiLength = input.int(14, title="QTI Period", minval=1)

// Custom function for Weighted Moving Average (WMA)
wma(src, length) =>
    sum = 0.0
    sumw = 0.0
    for i = 0 to length - 1
        weight = (length * (length + 1)) / 2 - (i * (i + 1)) / 2
        sum := sum + src[i] * weight
        sumw := sumw + weight
    sum / sumw

// Calculate QTI using WMA
qti(src, length) =>
    wma(src - wma(src, length), length)

qtiValue = qti(close, qtiLength)

// Calculate Accelerator Oscillator (AO)
aoLength = input.int(5, title="AO Smoothing", minval=1)
aoSMA = input.int(5, title="AO SMA", minval=1)

ao = ta.sma(ta.sma((high + low) / 2 - ta.sma((high + low) / 2, aoLength), aoSMA), aoSMA)

// Plot QTI and AO
plot(qtiValue, color=color.blue, title="QTI")
plot(ao, color=color.red, title="Accelerator Oscillator")

// Plot zero line for AO
hline(0, "Zero Line", color=color.gray)

// Plot signals
plotshape(qtiValue > 0 and qtiValue[1] <= 0 ? true : na, title="Buy Signal", location=location.belowbar, color=color.green, style=shape.labelup, text="Buy")
plotshape(qtiValue < 0 and qtiValue[1] >= 0 ? true : na, title="Sell Signal", location=location.abovebar, color=color.red, style=shape.labeldown, text="Sell")

// Alerts
alertcondition(qtiValue > 0 and qtiValue[1] <= 0, title="QTI Buy Signal", message="QTI Buy Signal")
alertcondition(qtiValue < 0 and qtiValue[1] >= 0, title="QTI Sell Signal", message="QTI Sell Signal")
