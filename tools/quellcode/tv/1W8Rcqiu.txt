//@version=5
indicator("Fast Trades - DeMarker + MA + Volatility Filter", overlay=true)

// Input settings
cciLength = input.int(14, title="CCI Length")
maLength = input.int(20, title="MA Length")
atrLength = input.int(14, title="ATR Length")
threshold = input.float(1.5, title="Threshold for DeMarker")
srcMA = input.source(close, title="MA Source")
volatilityThreshold = input.float(0.5, title="Volatility Threshold")

// Calculate DeMarker
highMax = ta.highest(high, cciLength)
lowMin = ta.lowest(low, cciLength)
demValue = (close - lowMin) / (highMax - lowMin)

// Calculate MA with selectable source
maValue = ta.sma(srcMA, maLength)

// Calculate ATR
atrValue = ta.atr(atrLength)

// Calculate Volatility
volatility = atrValue / ta.sma(atrValue, atrLength)

// Plot DeMarker, MA, ATR and Volatility
plot(demValue, title="DeMarker", color=color.blue, linewidth=2, display=display.none)
plot(maValue, title="MA", color=color.orange, linewidth=2, display=display.none)
plot(atrValue, title="ATR", color=color.green, linewidth=2, display=display.none)
plot(volatility, title="Volatility", color=color.purple, linewidth=2, display=display.none)

// Determine Buy and Sell Conditions
buyCondition = demValue < threshold and close > maValue + atrValue and volatility > volatilityThreshold
sellCondition = demValue > 1 - threshold and close < maValue - atrValue and volatility > volatilityThreshold

// Filter for stronger signals
var float lastBuyPrice = na
var float lastSellPrice = na
if buyCondition
    lastBuyPrice := close
if sellCondition
    lastSellPrice := close

buySignal = buyCondition and (lastBuyPrice[1] == na or close > lastBuyPrice[1])
sellSignal = sellCondition and (lastSellPrice[1] == na or close < lastSellPrice[1])

// Plot Buy and Sell Signals
plotshape(series=buySignal, title="-Buy-Signal-", style=shape.labelup, location=location.belowbar, color=color.green, text="BUY", textcolor=color.black, size=size.tiny)
plotshape(series=sellSignal, title="-Sell-Signal-", style=shape.labeldown, location=location.abovebar, color=color.red, text="SELL", textcolor=color.black, size=size.tiny)

// Alerts
alertcondition(buySignal, title="Buy Signal", message="Buy Signal: DeMarker crosses below threshold and price is above MA + ATR with high volatility.")
alertcondition(sellSignal, title="Sell Signal", message="Sell Signal: DeMarker crosses above threshold and price is below MA - ATR with high volatility.")
