//@version=5
indicator(title="-Quick Stochastic Indicator-(QSI)-", shorttitle="-Quick Stochastic Indicator-(QSI)-", overlay=true)

// Inputparameter
stoch_k_period = input.int(14, title="-Stochastic-K-Period-14-", minval=1)
stoch_d_period = input.int(3, title="-Stochastic-D-Period-3-", minval=1)
smooth_k = input.int(3, title="-Smooth-%K-3-", minval=1, maxval=50)
smooth_d = input.int(3, title="-Smooth-%D-3-", minval=1, maxval=50)

// Berechnung des Stochastic Oscillators
h = ta.highest(high, stoch_k_period)
l = ta.lowest(low, stoch_k_period)
k = ta.sma(100 * (close - l) / (h - l), smooth_k)
d = ta.sma(k, smooth_d)

// Bestimmen von Hoch- und Tiefpunkten
is_high = ta.highest(high, stoch_k_period) == high
is_low = ta.lowest(low, stoch_k_period) == low

// Kauf- und Verkaufssignale nur an Hoch- und Tiefpunkten
buy_signal = ta.crossover(k, d) and is_low
sell_signal = ta.crossunder(k, d) and is_high

// Plots für Kauf- und Verkaufssignale
plotshape(series=buy_signal, title="-Buy-Signal-", location=location.belowbar, color=color.green, textcolor=color.black, style=shape.labelup, text="BUY", size=size.tiny)
plotshape(series=sell_signal, title="-Sell-Signal-", location=location.abovebar, color=color.red, textcolor=color.black, style=shape.labeldown, text="SELL", size=size.tiny)

// Linien für den Stochastic Oscillator
plot(k, title="-%K-", color=color.aqua, linewidth=2)
plot(d, title="-%D-", color=color.red, linewidth=3)
