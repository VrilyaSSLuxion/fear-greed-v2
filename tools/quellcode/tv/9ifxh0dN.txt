toSize(string s) =>
    switch s
        "tiny"   => size.tiny
        "small"  => size.small
        "normal" => size.normal
        "large"  => size.large
        "huge"   => size.huge
        => size.normal

toLabelStyle(string s) =>
    switch s
        "Label Up"     => label.style_label_up
        "Label Down"   => label.style_label_down
        "Label Left"   => label.style_label_left
        "Label Right"  => label.style_label_right
        "Label Center" => label.style_label_center
        "Pfeil Up"     => label.style_arrowup
        "Pfeil Down"   => label.style_arrowdown
        "Dreieck Up"   => label.style_triangleup
        "Dreieck Down" => label.style_triangledown
        "Kreis"        => label.style_circle
        "Kreuz"        => label.style_cross
        "Diamant"      => label.style_diamond
        "Flagge"       => label.style_flag
        "Quadrat"      => label.style_square
        "Nur Text"     => label.style_none
        => label.style_none

toLineStyle(string s) =>
    switch s
        "DASHED" => line.style_dashed
        "DOTTED" => line.style_dotted
        => line.style_solid

detectSignals(float osc, float upper, float lower) =>
    var int   nextSig = 0
    var float seqTop  = -105.0
    var float seqBot  = 5.0
    var float peakTop = na
    var float peakBot = na
    var bool  inTop   = false
    var bool  inBot   = false
    bool trigTop = false
    bool trigBot = false

    if not na(osc)
        if osc > upper
            if not inTop
                inTop   := true
                peakTop := osc
            else if osc > peakTop
                peakTop := osc
        else if inTop
            if not na(peakTop) and peakTop > upper
                if nextSig == 0 or nextSig == 1 or (nextSig == -1 and peakTop > seqTop)
                    trigTop := true
                    seqTop  := peakTop
                    nextSig := -1
            inTop   := false
            peakTop := na

        if osc < lower
            if not inBot
                inBot   := true
                peakBot := osc
            else if osc < peakBot
                peakBot := osc
        else if inBot
            if not na(peakBot) and peakBot < lower
                if nextSig == 0 or nextSig == -1 or (nextSig == 1 and peakBot < seqBot)
                    trigBot := true
                    seqBot  := peakBot
                    nextSig := 1
            inBot   := false
            peakBot := na
    [trigTop, trigBot]

        if na(liveExt) or high > liveExt
            liveExt := high
            liveBar := bar_index
            liveH   := high
            liveL   := low
            redraw  := true
    if not na(confirmedBar) and confirmedBar == liveBar
        clearLive()
        redraw := false
    if redraw and armDir != 0
        clearLive()
        float riskV  = liveH - liveL
        float entryV = armDir == 1 ? liveH : liveL
        if liveArrow
            if armDir == 1
                array.push(liveLbls, label.new(liveBar, liveL, longText, color=color.new(longBgColor, longBgTransp), style=toLabelStyle(longStyle), size=toSize(longSize), textcolor=liveColL, force_overlay=true))

                array.push(liveLbls, label.new(liveBar, liveH, shortText, color=color.new(shortBgColor, shortBgTransp), style=toLabelStyle(shortStyle), size=toSize(shortSize), textcolor=liveColS, force_overlay=true))
        if liveLinesOn
            color hiC = color.new(armDir == 1 ? longHighColor : shortHighColor, liveDim)
            color loC = color.new(armDir == 1 ? longLowColor  : shortLowColor,  liveDim)
            array.push(liveLines, line.new(liveBar, liveH, bar_index, liveH, color=hiC, width=lineWidth, style=lineStyleStr, force_overlay=true))
            array.push(liveLines, line.new(liveBar, liveL, bar_index, liveL, color=loC, width=lineWidth, style=lineStyleStr, force_overlay=true))
        if liveTpOn and tpMaster and riskV > 0
            for i = 0 to array.size(lvMult) - 1
                if array.get(lvOn, i)
                    float lvl = entryV + armDir * array.get(lvMult, i) * riskV
                    array.push(liveLines, line.new(liveBar, lvl, bar_index, lvl, color=color.new(array.get(lvCol, i), liveDim), width=array.get(lvWid, i), style=toLineStyle(array.get(lvSty, i)), force_overlay=true))
