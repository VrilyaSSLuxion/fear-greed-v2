//@version=5
indicator("-VrilyaSS-Keltner Strategy with RSI and Swing Points-", overlay=true)

// Input parameters for Keltner Channel
length = input.int(25, title="EMA Length")
atr_length = input.int(30, title="ATR Length")
mult = input.float(2, title="ATR Multiplier")

// Input parameters for RSI
rsi_source = input.source(close, title="RSI Source")
rsi_length = input.int(14, title="RSI Length")

// Input parameters for Swing Points
swing_length = input.int(5, title="Swing Length")

// Calculate Keltner Channel
ema = ta.ema(close, length)
atr = ta.atr(atr_length)
upper_band = ema + mult * atr
lower_band = ema - mult * atr

// Plot Keltner Channel
plot(upper_band, color=color.red, linewidth = 3, title="Upper Keltner Band")
plot(ema, color=color.white, linewidth = 2, title="EMA")
plot(lower_band, color=color.green, linewidth = 3, title="Lower Keltner Band")

// Calculate RSI
rsi = ta.rsi(rsi_source, rsi_length)

// Calculate Swing Highs and Lows
swing_high = ta.valuewhen(ta.highestbars(high, swing_length) == 0, high, 0)
swing_low = ta.valuewhen(ta.lowestbars(low, swing_length) == 0, low, 0)

// Define Buy and Sell Conditions
buy_condition = close < lower_band and rsi < 30 and ta.lowestbars(low, swing_length) == 0
sell_condition = close > upper_band and rsi > 70 and ta.highestbars(high, swing_length) == 0

// Plot Buy and Sell Signals
plotshape(series=buy_condition, location=location.belowbar, color=color.green, textcolor=color.black, style=shape.labelup, text="B", title="Buy Signal", size=size.small)
plotshape(series=sell_condition, location=location.abovebar, color=color.red, textcolor=color.black, style=shape.labeldown, text="S", title="Sell Signal", size=size.small)

// Alert Conditions
alertcondition(buy_condition, title="Buy Signal Alert", message="Keltner Channel Buy Signal")
alertcondition(sell_condition, title="Sell Signal Alert", message="Keltner Channel Sell Signal")

// General Buy/Sell Signal Alert
alertcondition(buy_condition or sell_condition, title="Buy/Sell Signal Alert", message="Keltner Channel Buy or Sell Signal")
