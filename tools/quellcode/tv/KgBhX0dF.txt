//@version=5
indicator(title="RSI and EMA Strategy", shorttitle="RSI-EMA", overlay=true)

// Input parameters
rsi_length = input.int(14, title="RSI Length")
ema_length = input.int(9, title="EMA Length")
oversold_level = input.int(30, title="Oversold Level")
overbought_level = input.int(70, title="Overbought Level")
mid_level = 50

// Checkbox inputs
enable_long_signals = input.bool(true, title="Enable Long Signals")
enable_short_signals = input.bool(true, title="Enable Short Signals")

// Calculate RSI
rsi_value = ta.rsi(close, rsi_length)

// Calculate EMA
ema_value = ta.ema(close, ema_length)

// Plot RSI and EMA
plot(rsi_value, title="RSI", color=color.blue)
plot(ema_value, title="EMA", color=color.red)

// Plot horizontal lines for oversold, overbought, and mid levels
hline(oversold_level, "Oversold", color=color.green)
hline(overbought_level, "Overbought", color=color.red)
hline(mid_level, "Mid Level", color=color.gray)

// Strategy conditions
long_condition = enable_long_signals and (ta.crossover(rsi_value, mid_level) or (ta.crossunder(rsi_value, mid_level) and rsi_value > mid_level))
short_condition = enable_short_signals and (ta.crossunder(rsi_value, mid_level) or (ta.crossover(rsi_value, mid_level) and rsi_value < mid_level))

// Plot buy and sell signals with larger triangles at crossover points
plotshape(series=long_condition, location=location.belowbar, color=color.green, style=shape.triangleup, title="Buy Signal", size=size.large, text="BUY")
plotshape(series=short_condition, location=location.abovebar, color=color.red, style=shape.triangledown, title="Sell Signal", size=size.large, text="SELL")

// Strategy alerts
alertcondition(long_condition, title="Buy Signal", message="Buy Signal on {{ticker}}!")
alertcondition(short_condition, title="Sell Signal", message="Sell Signal on {{ticker}}!")
