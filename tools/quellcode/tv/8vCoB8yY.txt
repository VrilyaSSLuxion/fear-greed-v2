oscRaw(simple int length, simple bool wait) =>
    float hh = ta.highest(high, length)
    float ll = ta.lowest(low, length)

    wait ? v[1] : v

dropLastSet() =>
    int freed = 0
    if array.size(setLines) > 0
        int delL = array.pop(setLines)
        int delB = array.pop(setLbls)
        int delA = array.pop(setArrows)
        int delE = array.pop(setEx)
        int delX = array.pop(setBoxes)
        array.pop(setSeq)
        if delL > 0
            for k = 1 to delL
                if array.size(allLines) > 0
                    line.delete(array.pop(allLines))
        if delB > 0
            for k = 1 to delB
                if array.size(allLabels) > 0
                    label.delete(array.pop(allLabels))
        if delA > 0
            for k = 1 to delA
                if array.size(allArrows) > 0
                    label.delete(array.pop(allArrows))
        if delX > 0
            for k = 1 to delX
                if array.size(allBoxes) > 0
                    box.delete(array.pop(allBoxes))
        if delE > 0
            for k = 1 to delE
                if array.size(allEx) > 0
                    label.delete(array.pop(allEx))
            if array.size(pExUsed) > 0
                for k = 0 to array.size(pExUsed) - 1
                    array.set(pExUsed, k, false)
            freed := delE
    freed

drawBox(int anchor, float cH, float cL, float cO, float cC, bool isLong, int rightBar) =>
    float top = boxSrc == "Körper (Open/Close)" ? math.max(cO, cC) : cH
    float bot = boxSrc == "Körper (Open/Close)" ? math.min(cO, cC) : cL
    float pad = boxPadPct > 0 ? (top - bot) * boxPadPct / 100.0 : 0.0
    int   l   = anchor - boxPadL
    int   r   = boxFollow ? math.max(bar_index, anchor + boxPadR) : anchor + boxPadR
    if r <= l
        r := l + 1

detectSignals(float osc, float upper, float lower) =>
    var int   nextSig = 0
    var float seqTop  = -105.0
    var float seqBot  = 5.0
    var float peakTop = na
    var float peakBot = na
    var bool  inTop   = false
    var bool  inBot   = false
    bool trigTop = false
    bool trigBot = false

    if not na(osc)
        if osc > upper
            if not inTop
                inTop   := true
                peakTop := osc
            else if osc > peakTop
                peakTop := osc
        else if inTop
            if not na(peakTop) and peakTop > upper
                if nextSig == 0 or nextSig == 1 or (nextSig == -1 and peakTop > seqTop)
                    trigTop := true
                    seqTop  := peakTop
                    nextSig := -1
            inTop   := false
            peakTop := na

        if osc < lower
            if not inBot
                inBot   := true
                peakBot := osc
            else if osc < peakBot
                peakBot := osc
        else if inBot
            if not na(peakBot) and peakBot < lower
                if nextSig == 0 or nextSig == -1 or (nextSig == 1 and peakBot < seqBot)
                    trigBot := true
                    seqBot  := peakBot
                    nextSig := 1
            inBot   := false
            peakBot := na
    [trigTop, trigBot]
