//@version=5
indicator("-VrilyaSS-EMA-Crossover-3-EMA+6-EMA-", overlay=true)

// Inputs
fastLength = input(3, title="-Fast-EMA-Length-3-")
slowLength = input(6, title="-Slow-EMA-Length-6-")
src = input(close, title="-Source-close-")

// Calculate exponential moving averages
fastEMA = ta.ema(src, fastLength)
slowEMA = ta.ema(src, slowLength)

// Plot exponential moving averages with thicker lines (linewidth=2)
plot(fastEMA, color=color.blue, title="-Fast-EMA-", linewidth=2)
plot(slowEMA, color=color.red, title="-Slow-EMA-", linewidth=2)

// Generate trading signals
longCondition = ta.crossover(fastEMA, slowEMA)
shortCondition = ta.crossunder(fastEMA, slowEMA)

// Plot arrows for buy/sell signals
plotshape(series=longCondition, title="-Buy-Signal-", location=location.belowbar, color=color.green, textcolor=color.black, style=shape.labelup, text="Buy", size=size.tiny)
plotshape(series=shortCondition, title="-Sell-Signal-", location=location.abovebar, color=color.red, textcolor=color.black, style=shape.labeldown, text="Sell", size=size.tiny)

// Alert conditions
alertcondition(longCondition, title="-Buy-Signal-VrilyaSS-EMA-Crossover-3-EMA+6-EMA-", message="-Buy-Fast-EMA-3-crossed above Slow-EMA-6-")
alertcondition(shortCondition, title="-Sell-Signal-VrilyaSS-EMA-Crossover-3-EMA+6-EMA-", message="-Sell-Fast-EMA-3-crossed below Slow-EMA-6-")

// Chart Title
var label titleLabel = na
if not na(titleLabel)
    label.delete(titleLabel)

showTitle = input(false, title="Show Chart Title")
if showTitle
    titleLabel := label.new(x=bar_index, y=high, text="-Moving-Average-Crossover-Indicator-", style=label.style_label_down, color=color.blue, textcolor=color.white, size=size.tiny)
