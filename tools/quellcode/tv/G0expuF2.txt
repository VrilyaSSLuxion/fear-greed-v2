toLog(series float value) =>
    float exponent = math.abs(value)
    float product  = math.log10(exponent + 0.0001) + 4
    float result   = exponent < 1e-8 ? 0 : value < 0 ? - product : product

fromLog(series float value) =>
    float exponent = math.abs(value)
    float product  = math.pow(10, exponent - 4) - 0.0001
    float result   = exponent < 1e-8 ? 0 : value < 0 ? - product : product

lineStyle(simple string styleString) =>
    switch styleString
        TY01 => line.style_solid
        TY02 => line.style_dashed
        TY03 => line.style_dotted

labelText(series float value, series float price, series bool showP, series bool showL, series bool showPct) =>
    string levelString = showPct ? str.tostring(value * 100, format.percent) : str.format("{0, number, #.###}", value)
    switch
        showP and showL => str.format("{0}({1})", levelString, str.tostring(price, format.mintick))
        showP           => str.format("({0})", str.tostring(price, format.mintick))
        showL           => levelString
        => ""

labelTimeCalc(series int leftTime, series int rightTime, series int finalRightTime, series string lblX, series bool extBars, series bool extRt) =>
    switch lblX
        LB01 => leftTime
        LB02 => int(math.avg(leftTime, extBars ? finalRightTime : extRt ? chart.right_visible_bar_time : rightTime))
        LB03 => extBars ? finalRightTime : extRt ? chart.right_visible_bar_time : rightTime

timeLabel(series int x, series float y, simple color col, series string style, series string size) =>
    var label tL = label.new(na, na, na, xloc.bar_time, yloc.price, color.new(col, 80), style, col, size)
    label.set_xy(tL, x, y)
    label.set_text(tL, str.format("{0}\n{1}", str.tostring(y, format.mintick), str.format_time(x, "dd/MM/yy @ HH:mm:ss")))

hiLoLine(series int x1, series float y1, series int x2, series float y2, simple color col, simple string sty, simple int w) =>
    var line hl = line.new(na, na, na, na, xloc.bar_time, extend.none, col, lineStyle(sty), w)
    line.set_xy1(hl, x1, y1)
    line.set_xy2(hl, x2, y2)

addToArray(id, simple int max, value) =>
    array.unshift(id, value)

addPivot_1(simple int len, array<pivotPoint_1> arr, simple bool addP = true) =>
    pivotPoint_1 p = array.get(arr, 0)
    int aL = math.max(p.bars, 1)
    int hB = ta.highestbars(len)
    int lB = ta.lowestbars(len)
    int ahB = ta.highestbars(aL)
    int alB = ta.lowestbars(aL)
    float aH = ta.highest(aL)
    float aLo = ta.lowest(aL)
    bool mP = p.bars >= len and addP and not (hB == 0 and high > p.end.y or lB == 0 and low < p.end.y)
    var int d = na
    d := switch
        mP => nz(d, ahB < alB ? 1 : -1) * -1
        hB + lB == 0 => d * -1
        hB == 0 => 1
        lB == 0 => -1
        => d
    if d != d[1] and time <= chart.right_visible_bar_time
        [hL, aHL, off, lc] = switch
            d > 0 => [high, aH, ahB, zzBullColorInput_1]
            => [low, aLo, alB, zzBearColorInput_1]
        [pY, bs] = switch
            mP => [aHL, math.abs(off)]
            => [hL, 0]
        point_1 np = point_1.new(pY, time[bs])
        pivotPoint_1 piv = pivotPoint_1.new(p.end, np, d, bars = bs)
        if showZZInput_1 and time >= chart.left_visible_bar_time
            piv.ln := line.new(p.end.x, p.end.y, time[bs], pY, xloc.bar_time, extend.none, lc, lineStyle(zzStyleInput_1), zzwidthInput_1)
        addToArray(arr, math.max(3, pivotInput_1 + 1), piv)

updatePivot_1(simple int len, array<pivotPoint_1> arr) =>
    if time <= chart.right_visible_bar_time
        pivotPoint_1 p = array.get(arr, 0)
        int d = p.dir
        float hL = d > 0 ? high : low
        p.bars += 1
        if hL * d > p.end.y * d
            p.end.y := hL
            p.end.x := time
            p.bars := 0
            if showZZInput_1 and time >= chart.left_visible_bar_time
                if na(p.ln)
                    color lc = d > 0 ? zzBullColorInput_1 : zzBearColorInput_1
                    p.ln := line.new(p.start.x, p.start.y, time, hL, xloc.bar_time, extend.none, lc, lineStyle(zzStyleInput_1), zzwidthInput_1)
            line.set_xy2(p.ln, time, hL)

    addPivot_1(len, pivs, addP)

    float hiPoint  = useLogScaleInput_2 ? toLog(hiPrice) : hiPrice
    float loPoint  = useLogScaleInput_2 ? toLog(loPrice) : loPrice
    float fibPrice = (isBull and invertFibInput_2 or not isBull and not invertFibInput_2) ? loPoint + ((hiPoint - loPoint) * value) : hiPoint - ((hiPoint - loPoint) * value)
    useLogScaleInput_2 ? fromLog(fibPrice) : fibPrice

        array.push(levels_2, val)

setLevels_2(series int lT, series int rT, series float hP, series float lP, series bool iB, series int fRT) =>

addPivot_2(simple int len, array<pivotPoint_2> arr, simple bool addP = true) =>
    pivotPoint_2 p = array.get(arr, 0)
    int aL = math.max(p.bars, 1)
    int hB = ta.highestbars(len)
    int lB = ta.lowestbars(len)
    int ahB = ta.highestbars(aL)
    int alB = ta.lowestbars(aL)
    float aH = ta.highest(aL)
    float aLo = ta.lowest(aL)
    bool mP = p.bars >= len and addP and not (hB == 0 and high > p.end.y or lB == 0 and low < p.end.y)
    var int d = na
    d := switch
        mP => nz(d, ahB < alB ? 1 : -1) * -1
        hB + lB == 0 => d * -1
        hB == 0 => 1
        lB == 0 => -1
        => d
    if d != d[1] and time <= chart.right_visible_bar_time
        [hL, aHL, off, lc] = switch
            d > 0 => [high, aH, ahB, zzBullColorInput_2]
            => [low, aLo, alB, zzBearColorInput_2]
        [pY, bs] = switch
            mP => [aHL, math.abs(off)]
            => [hL, 0]
        point_2 np = point_2.new(pY, time[bs])
        pivotPoint_2 piv = pivotPoint_2.new(p.end, np, d, bars = bs)
        if showZZInput_2 and time >= chart.left_visible_bar_time
            piv.ln := line.new(p.end.x, p.end.y, time[bs], pY, xloc.bar_time, extend.none, lc, lineStyle(zzStyleInput_2), zzwidthInput_2)
        addToArray(arr, math.max(3, pivotInput_2 + 1), piv)

updatePivot_2(simple int len, array<pivotPoint_2> arr) =>
    if time <= chart.right_visible_bar_time
        pivotPoint_2 p = array.get(arr, 0)
        int d = p.dir
        float hL = d > 0 ? high : low
        p.bars += 1
        if hL * d > p.end.y * d
            p.end.y := hL
            p.end.x := time
            p.bars := 0
            if showZZInput_2 and time >= chart.left_visible_bar_time
                if na(p.ln)
                    color lc = d > 0 ? zzBullColorInput_2 : zzBearColorInput_2
                    p.ln := line.new(p.start.x, p.start.y, time, hL, xloc.bar_time, extend.none, lc, lineStyle(zzStyleInput_2), zzwidthInput_2)
            line.set_xy2(p.ln, time, hL)

zigZag_2(simple int len, simple bool addP = true) =>
    var array<pivotPoint_2> pivs = array.new<pivotPoint_2>(1, pivotPoint_2.new(point_2.new(close), point_2.new(close)))
    updatePivot_2(len, pivs)
    addPivot_2(len, pivs, addP)
    pivs

findHiLo_2(simple int nth) =>
    int lX = na
    float lY = na
    int rX = na
    float rY = na
    if isZigZag_2
        array<pivotPoint_2> pa = zigZag_2(lengthInput_2, addPivotInput_2)
        if barstate.islast and array.size(pa) > nth
            pivotPoint_2 pv = array.get(pa, nth)
            lX := pv.start.x
            lY := pv.start.y
            rX := pv.end.x
            rY := pv.end.y
            if showZZInput_2
                line.set_style(pv.ln, lineStyle(trendStyleInput_2))
    [lX, lY, rX, rY]
