//@version=5
indicator("Combined Indicator", overlay=false)

// Input-Parameter für Indikatoren
length = input.int(14, title="Length for DeMarker, RSI, Stochastic")
src = input(close, title="Source")
fastLength = input.int(12, title="MACD Fast Length")
slowLength = input.int(26, title="MACD Slow Length")
signalSmoothing = input.int(9, title="MACD Signal Smoothing")
aoShortLength = input.int(5, title="Awesome Oscillator Short Length")
aoLongLength = input.int(34, title="Awesome Oscillator Long Length")

// Berechnung der Indikatoren
// DeMarker
dmHigh = ta.highest(src, length)
dmLow = ta.lowest(src, length)
dmDiff = dmHigh - dmLow
demarker = na(dmDiff) ? na : (dmHigh == dmLow ? 0 : (dmHigh - src) / dmDiff)

// RSI
rsi = ta.rsi(src, length)

// Stochastic Oscillator
float k = na
float d = na
var stochTuple = ta.stoch(src, length, 3, 3)
k := stochTuple[1]
d := stochTuple[2]

// MACD
[macdLine, signalLine, _] = ta.macd(src, fastLength, slowLength, signalSmoothing)

// Awesome Oscillator
float aoShort = na
float aoLong = na
aoShort := ta.sma(close, aoShortLength)
aoLong := ta.sma(close, aoLongLength)
float ao = aoShort - aoLong

// Logik für Buy- und Sell-Signale
bool buySignal = (demarker > 0.7 and rsi < 30 and k > d and macdLine > signalLine and ao > 0)
bool sellSignal = (demarker < 0.3 and rsi > 70 and k < d and macdLine < signalLine and ao < 0)

// Plot der Signale
plotshape(series=buySignal, title="Buy Signal", location=location.belowbar, color=color.green, style=shape.labelup, text="BUY")
plotshape(series=sellSignal, title="Sell Signal", location=location.abovebar, color=color.red, style=shape.labeldown, text="SELL")
