//@version=5
indicator("Volume Delta with Filters", overlay=true)

// Input settings
smaLength = input.int(50, title="SMA Length")
macdShortLength = input.int(12, title="MACD Short Length")
macdLongLength = input.int(26, title="MACD Long Length")
macdSignalSmoothing = input.int(9, title="MACD Signal Smoothing")
atrLength = input.int(14, title="ATR Length")
atrMultiplier = input.float(1.5, title="ATR Multiplier")
volumeDeltaThreshold = input.int(100000, title="Volume Delta Threshold")
maxConsecutiveCandles = input.int(3, title="Max Consecutive Bullish/Bearish Candles")
enableBuySignal = input.bool(true, title="Enable Buy Signal")
enableSellSignal = input.bool(true, title="Enable Sell Signal")

// Calculate Volume Delta
volumeDelta = ta.change(volume)
deltaVolumePositive = volumeDelta > volumeDeltaThreshold
deltaVolumeNegative = volumeDelta < -volumeDeltaThreshold

// Calculate SMA Trendline
sma = ta.sma(close, smaLength)

// Calculate MACD
[macdLine, signalLine, _] = ta.macd(close, macdShortLength, macdLongLength, macdSignalSmoothing)
macdHist = macdLine - signalLine

// Calculate ATR and ATR Range Filter
atr = ta.atr(atrLength)
atrRangeFilter = atr * atrMultiplier

// Candle Counting for Filtering
bullishCount = ta.barssince(close < open)
bearishCount = ta.barssince(close > open)

// Buy and Sell Conditions with Filters
buyCondition = enableBuySignal and deltaVolumePositive and macdHist > 0 and close > sma and atr < atrRangeFilter and bullishCount <= maxConsecutiveCandles
sellCondition = enableSellSignal and deltaVolumeNegative and macdHist < 0 and close < sma and atr < atrRangeFilter and bearishCount <= maxConsecutiveCandles

// Plot Buy and Sell Signals
plotshape(series=buyCondition, title="Buy Signal", location=location.belowbar, color=color.green, style=shape.labelup, text="BUY")
plotshape(series=sellCondition, title="Sell Signal", location=location.abovebar, color=color.red, style=shape.labeldown, text="SELL")

// Alerts
alertcondition(buyCondition, title="Buy Signal", message="Buy Signal detected")
alertcondition(sellCondition, title="Sell Signal", message="Sell Signal detected")
