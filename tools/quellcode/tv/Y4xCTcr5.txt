//@version=5
indicator("-VrilyaSS-RSI Bounce Buy/Sell Signals with Trend Reversals-", overlay=true)

// Input parameters for RSI
rsi_source = input.source(close, title="RSI Source")
rsi_length = input.int(14, title="RSI Length")

// Calculate RSI
rsi = ta.rsi(rsi_source, rsi_length)

// Define the swing high and low periods
swing_period = input.int(3, title="Swing Period")

// Calculate Swing Highs and Lows
swing_high = ta.highest(high, swing_period)
swing_low = ta.lowest(low, swing_period)

// Define the states
var bool was_below_50 = na
var bool was_above_50 = na

// Track the state of RSI crossing above or below 50
if (rsi < 50)
    was_below_50 := true
    was_above_50 := false
if (rsi > 50)
    was_above_50 := true
    was_below_50 := false

// Define Buy and Sell Conditions with Swing High/Low Confirmation
buy_condition = was_below_50[1] and rsi > 50 and low == swing_low
sell_condition = was_above_50[1] and rsi < 50 and high == swing_high

// Plot Buy and Sell Signals
plotshape(series=buy_condition, location=location.belowbar, color=color.green, textcolor=color.black, style=shape.labelup, text="B", title="Buy Signal", size=size.small)
plotshape(series=sell_condition, location=location.abovebar, color=color.red, textcolor=color.black, style=shape.labeldown, text="S", title="Sell Signal", size=size.small)

// Alert Conditions
alertcondition(buy_condition, title="Buy Signal Alert", message="RSI Bounce Buy Signal")
alertcondition(sell_condition, title="Sell Signal Alert", message="RSI Bounce Sell Signal")
