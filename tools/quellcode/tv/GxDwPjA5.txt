//@version=5
indicator(title="-VrilyaSS-Stochastic+DIVERGENCE+3X-BANDS-", shorttitle="-VrilyaSS-Stochastic+DIVERGENCE+3X-BANDS-", format=format.price, precision=2, timeframe="", timeframe_gaps=true)
periodK = input.int(21, title="-%K-Length-(A5)-(B21)-(C30)-(D38)-default-14-", minval=1)
smoothK = input.int(5, title="-%K-Smoothing-(A3)-(B5)-(C1)-(D4)-default-1-", minval=1)
periodD = input.int(5, title="-%D-Smoothing-(A2)-(B5)-(C3)-(D5)-default-3-", minval=1)
k = ta.sma(ta.stoch(close, high, low, periodK), smoothK)
d = ta.sma(k, periodD)
plot(k, title="-%K-Color-light-blue-16-Solid-Line-2-100%-", linewidth=2, color=#00bcd4)
plot(d, title="-%D-Color-red-11-Solid-Line-3-100%-", linewidth=3, color=#f23645)

h0 = hline(99, "-UPPER TOP-Band---TOP--(99)--90--(90--89,11)---Color-11-Solid Line-3-100%-", linestyle=hline.style_solid, linewidth=3, color=#f23645)
hline(89, "-UPPER TOP-Band---MIDDLE--(89)--80--(79--78,6)---Color-1-Solid-Line-2-100%-", linestyle=hline.style_solid, linewidth=2, color=color.new(color.white, 0))
h1 = hline(79, "-UPPER TOP-Band---BOTTOM--(79)--70--(70--bzw-70,20-The Middle of 61,8 and 78,6)---Color-16-Solid-Line-2-100%-", linestyle=hline.style_solid, linewidth=2, color=color.new(color.aqua, 0)) 
fill(h0, h1, color=color.new(color.red, 85), title="-Background-UPPER TOP-Band---Color-11-15%-")

h0_2 = hline(56, "-MIDDLE NEUTRAL-Band---TOP--(56)--(62--61,8)---Color-11-Solid-Line-1-100%-", linestyle=hline.style_solid, linewidth=1, color=#f23645)
hline(50, "-MIDDLE NEUTRAL-Band---MIDDLE--(50)--(50--50,0)---Color-1-Solid-Line-3-100%-", linestyle=hline.style_solid, linewidth=3, color=color.new(color.white, 0))
h1_2 = hline(44, "-MIDDLE NEUTRAL-Band---BOTTOM--(44)--(38--38,2)---Color-14-Solid-Line-1-100%-", linestyle=hline.style_solid, linewidth=1, color=color.new(color.green, 0))
fill(h0_2, h1_2, color=color.new(color.gray, 85), title="-Background-MIDDLE NEUTRAL-Band---NTZ-MIDDLE-BAND-NoTradeZone-BAND---MIDDLE NEUTRAL CONSOLIDATION---AVOID THE MOST TIMES AND ONLY USE IT SOMETIMES FOR SUPPORT AND-OR BREAKTROUGH AS A CONFIRMATION-THE MOST TIMES THIS IS A ABSOLUTE NO-TRADE-ZONE AND SHOULD BE AVOIDED TO TRADE-Band---USE IT AS A BREAK-OUTS-LEVEL AND FOR BOUNCES AS A LEVEL FOR SUPPORT, AND-OR AS A BREAK-DOWN-LEVEL AND RESISTANCES LEVEL FOR REJECTIONS---Color-5-15%-")

h0_3 = hline(21, "-LOWER BOTTOM-Band---TOP--(21)--30---Color-16-Solid-Line-2-100%-", linestyle=hline.style_solid, linewidth=2, color=color.new(color.aqua, 0))
hline(11, "-LOWER BOTTOM-Band---MIDDLE--(11)--20---Color-1-Solid-Line-2-100%-", linestyle=hline.style_solid, linewidth=2, color=color.new(color.white, 0))
h1_3 = hline(1, "-LOWER BOTTOM-Band---BOTTOM--(1)--10---Color-14-Solid-Line-3-100%-", linestyle=hline.style_solid, linewidth=3, color=color.new(color.green, 0))
fill(h0_3, h1_3, color=color.new(color.green, 85), title="-Background-LOWER BOTTOM-Band---Color-14-15%-")

// Divergence Detection Logic
lbR = input.int(title="-Pivot Lookback Right-5-", defval=5, display=display.data_window)
lbL = input.int(title="-Pivot Lookback Left-5-", defval=5, display=display.data_window)
rangeUpper = input.int(title="-Max of Lookback Range-60-", defval=60, display=display.data_window)
rangeLower = input.int(title="-Min of Lookback Range-5-", defval=5, display=display.data_window)
plotBull = input.bool(title="-Plot Bullish-true-", defval=true, display=display.data_window)
plotHiddenBull = input.bool(title="-Plot Hidden-Bullish-true-", defval=true, display=display.data_window)
plotBear = input.bool(title="-Plot Bearish-true-", defval=true, display=display.data_window)
plotHiddenBear = input.bool(title="-Plot Hidden-Bearish-true-", defval=true, display=display.data_window)

bullColor = input.color(title="-Regular-Bullish-Label-Color-", defval=color.green, group="Regular Bullish", inline="color")
bullTextColor = input.color(title="-Regular-Bullish-Label-Text-Color-", defval=color.white, group="Regular Bullish", inline="textColor")

hiddenBullColor = input.color(title="-Hidden-Bullish-Label-Color-", defval=color.new(color.green, 50), group="Hidden Bullish", inline="color")
hiddenBullTextColor = input.color(title="-Hidden-Bullish-Label-Text-Color-", defval=color.white, group="Hidden Bullish", inline="textColor")

bearColor = input.color(title="-Regular-Bearish-Label-Color-", defval=color.red, group="Regular Bearish", inline="color")
bearTextColor = input.color(title="-Regular-Bearish-Label-Text-Color-", defval=color.white, group="Regular Bearish", inline="textColor")

hiddenBearColor = input.color(title="-Hidden-Bearish-Label-Color-", defval=color.new(color.red, 50), group="Hidden Bearish", inline="color")
hiddenBearTextColor = input.color(title="-Hidden-Bearish-Label-Text-Color-", defval=color.white, group="Hidden Bearish", inline="textColor")

noneColor = color.new(color.white, 100)

// Detect pivots
plFound = na(ta.pivotlow(k, lbL, lbR)) ? false : true
phFound = na(ta.pivothigh(k, lbL, lbR)) ? false : true
_inRange(cond) =>
    bars = ta.barssince(cond == true)
    rangeLower <= bars and bars <= rangeUpper

// Regular Bullish Divergence
oscHL = k[lbR] > ta.valuewhen(plFound, k[lbR], 1) and _inRange(plFound[1])
priceLL = low[lbR] < ta.valuewhen(plFound, low[lbR], 1)
bullCondAlert = priceLL and oscHL and plFound
bullCond = plotBull and bullCondAlert

plot(plFound ? k[lbR] : na, offset=-lbR, title="-Regular Bullish-", linewidth=2, color=(bullCond ? bullColor : noneColor), display=display.pane)

plotshape(bullCond ? k[lbR] : na, offset=-lbR, title="-Regular Bullish-Label-", text=" Bull ", style=shape.labelup, location=location.absolute, color=bullColor, textcolor=bullTextColor)

// Hidden Bullish Divergence
oscLL = k[lbR] < ta.valuewhen(plFound, k[lbR], 1) and _inRange(plFound[1])
priceHL = low[lbR] > ta.valuewhen(plFound, low[lbR], 1)
hiddenBullCondAlert = priceHL and oscLL and plFound
hiddenBullCond = plotHiddenBull and hiddenBullCondAlert

plot(plFound ? k[lbR] : na, offset=-lbR, title="-Hidden Bullish-", linewidth=2, color=(hiddenBullCond ? hiddenBullColor : noneColor), display=display.pane)

plotshape(hiddenBullCond ? k[lbR] : na, offset=-lbR, title="-Hidden Bullish-Label-", text=" H Bull ", style=shape.labelup, location=location.absolute, color=hiddenBullColor, textcolor=hiddenBullTextColor)

// Regular Bearish Divergence
oscLH = k[lbR] < ta.valuewhen(phFound, k[lbR], 1) and _inRange(phFound[1])
priceHH = high[lbR] > ta.valuewhen(phFound, high[lbR], 1)
bearCondAlert = priceHH and oscLH and phFound
bearCond = plotBear and bearCondAlert

plot(phFound ? k[lbR] : na, offset=-lbR, title="-Regular Bearish-", linewidth=2, color=(bearCond ? bearColor : noneColor), display=display.pane)

plotshape(bearCond ? k[lbR] : na, offset=-lbR, title="-Regular Bearish-Label-", text=" Bear ", style=shape.labeldown, location=location.absolute, color=bearColor, textcolor=bearTextColor)

// Hidden Bearish Divergence
oscHH = k[lbR] > ta.valuewhen(phFound, k[lbR], 1) and _inRange(phFound[1])
priceLH = high[lbR] < ta.valuewhen(phFound, high[lbR], 1)
hiddenBearCondAlert = priceLH and oscHH and phFound
hiddenBearCond = plotHiddenBear and hiddenBearCondAlert

plot(phFound ? k[lbR] : na, offset=-lbR, title="-Hidden Bearish-", linewidth=2, color=(hiddenBearCond ? hiddenBearColor : noneColor), display=display.pane)

plotshape(hiddenBearCond ? k[lbR] : na, offset=-lbR, title="-Hidden Bearish-Label-", text=" H Bear ", style=shape.labeldown, location=location.absolute, color=hiddenBearColor, textcolor=hiddenBearTextColor)

alertcondition(bullCondAlert, title='Regular Bullish Divergence', message="Found a new Regular Bullish Divergence, Pivot Lookback Right number of bars to the left of the current bar")
alertcondition(hiddenBullCondAlert, title='Hidden Bullish Divergence', message='Found a new Hidden Bullish Divergence, Pivot Lookback Right number of bars to the left of the current bar')
alertcondition(bearCondAlert, title='Regular Bearish Divergence', message='Found a new Regular Bearish Divergence, Pivot Lookback Right number of bars to the left of the current bar')
alertcondition(hiddenBearCondAlert, title='Hidden Bearish Divergence', message='Found a new Hidden Bearish Divergence, Pivot Lookback Right number of bars to the left of the current bar')
